    63→function translateEventName(title: string, lang: string): string {
    64→  if (lang !== 'fr') return title;
    65→  const lower = title.toLowerCase();
    66→  // Try longest matches first
    67→  const sortedKeys = Object.keys(FRENCH_EVENT_TRANSLATIONS).sort((a, b) => b.length - a.length);
    68→  let result = title;
    69→  for (const en of sortedKeys) {
    70→    if (lower.includes(en)) {
    71→      // Case-insensitive replace
    72→      const regex = new RegExp(en, 'gi');
    73→      result = result.replace(regex, FRENCH_EVENT_TRANSLATIONS[en]);
    74→    }
    75→  }
    76→  return result;
    77→}

   180→function getImpactedPairs(title: string): string[] {
   181→  const t = title.toLowerCase();
   182→  const pairs: string[] = [];
   183→
   184→  // Currency-based
   185→  if (t.includes('usd') || t.includes('dollar') || t.includes('fed') || t.includes('fomc') || t.includes('nfp') || t.includes('non-farm') || t.includes('nonfarm') || t.includes('cpi') || t.includes('consumer price') || t.includes('gdp') || t.includes('rate decision') || t.includes('interest rate') || t.includes('jobless') || t.includes('claims') || t.includes('ppi') || t.includes('retail sales') || t.includes('consumer confidence') || t.includes('ism')) {
   186→    pairs.push('EUR/USD', 'XAU/USD');
   187→    if (t.includes('stock') || t.includes('fomc') || t.includes('rate') || t.includes('fed')) {
   188→      pairs.push('US30', 'US100');
   189→    }
   190→  }
   191→  if (t.includes('eur') || t.includes('euro') || t.includes('ecb') || t.includes('bce')) {
   192→    if (!pairs.includes('EUR/USD')) pairs.push('EUR/USD');
   193→    pairs.push('EUR/GBP');
   194→  }
   195→  if (t.includes('gbp') || t.includes('pound') || t.includes('boe') || t.includes('bda') || t.includes('bank of england')) {
   196→    if (!pairs.includes('GBP/USD')) pairs.push('GBP/USD');
   197→    pairs.push('EUR/GBP');
   198→  }
   199→  if (t.includes('jpy') || t.includes('yen') || t.includes('boj') || t.includes('japan') || t.includes('nikkei')) {
   200→    pairs.push('USD/JPY');
   201→  }
   202→  if (t.includes('gold') || t.includes('xau') || t.includes('oil') || t.includes('crude') || t.includes('commodit')) {
   203→    if (!pairs.includes('XAU/USD')) pairs.push('XAU/USD');
   204→  }
   205→  if (t.includes('stock') || t.includes('dow') || t.includes('sp ') || t.includes('s&p') || t.includes('earnings')) {
   206→    if (!pairs.includes('US30')) pairs.push('US30');
   207→    if (!pairs.includes('US100')) pairs.push('US100');
   208→  }
   209→  if (t.includes('nasdaq') || t.includes('tech') || t.includes('ai ')) {
   210→    if (!pairs.includes('US100')) pairs.push('US100');
   211→  }
   212→
   213→  // If nothing matched, default to major USD pairs
   214→  if (pairs.length === 0) {
   215→    pairs.push('EUR/USD', 'XAU/USD');
   216→  }
   217→
   218→  return [...new Set(pairs)]; // Deduplicate
   219→}

   302→async function fetchCalendarRSS(): Promise<any[]> {
   303→  const results: any[] = [];
   304→
   305→  const feedResults = await Promise.allSettled(
   306→    ECONOMIC_CALENDAR_FEEDS.map(feed =>
   307→      fetch(`https://www.investing.com/rss/news_${feed.id}.rss`, {
   308→        headers: { 'User-Agent': 'Mozilla/5.0 (compatible; DONCIEL-TM/1.0)' },
   309→        signal: AbortSignal.timeout(10000),
   310→      }).then(res => {
   311→        if (!res.ok) return [];
   312→        return res.text().then(xml => {
   313→          const parsed = parser.parse(xml);
   314→          const items = parsed?.rss?.channel?.item;
   315→          const list = Array.isArray(items) ? items : items ? [items] : [];
   316→          return list.filter((item: any) => {
   317→            const title = (item.title || '').toLowerCase();
   318→            return feed.keywords.some(kw => title.includes(kw)) && isEconomicEvent(title);
   319→          }).map((item: any) => ({ item, feed }));
   320→        });
   321→      }).catch(() => [])
   322→    )
   323→  );
   324→
   325→  for (const result of feedResults) {
   326→    if (result.status === 'fulfilled' && Array.isArray(result.value)) {
   327→      results.push(...result.value);
   328→    }
   329→  }
   330→
   331→  return results;
   332→}

   334→function parseCalendarFromRSS(rssItems: any[], lang: string, period: string): {
   335→  events: any[];
   336→  highImpactCount: number;
   337→  nextHighImpact: any | null;
   338→  updatedAt: string;
   339→} {
   340→  const isFr = lang === 'fr';
   341→  const now = new Date();
   342→  const isWeek = period === 'week';
   343→
   344→  const events = rssItems
   345→    .map(({ item }: any) => {
   346→      const title = item.title || '';
   347→      const pubDate = item.pubDate || '';
   348→      const impact = classifyImpact(title);
   349→      const { currency, flag } = detectCurrency(title);
   350→
   351→      // Try to extract date from pubDate
   352→      let eventDate: Date | null = null;
   353→      if (pubDate) {
   354→        try {
   355→          const d = new Date(pubDate);
   356→          if (!isNaN(d.getTime())) eventDate = d;
   357→        } catch {}
   358→      }
   359→      if (!eventDate) eventDate = now;
   360→
   361→      const eventName = translateEventName(title, lang);
   362→
   363→      return {
   364→        date: eventDate.toISOString().split('T')[0],
   365→        time: eventDate.toISOString().split('T')[1]?.substring(0, 5) || '',
   366→        currency,
   367→        impact,
   368→        event: eventName,
   369→        actual: null,
   370→        forecast: null,
   371→        previous: null,
   372→        country: flag,
   373→        interpretation: getEventInterpretation(title, lang),
   374→        direction: getEventDirection(title),
   375→        impactedPairs: getImpactedPairs(title),
   376→      };
   377→    })
   378→    .filter((evt: any) => {
   379→      // Filter by period
   380→      if (!isWeek) {
   381→        return evt.date === now.toISOString().split('T')[0];
   382→      }
   383→      // Week: for RSS, include ALL events without strict date filtering
   384→      // since RSS feeds mostly return recent/current events anyway
   385→      // But still broaden the range to include the whole current week
   386→      const dayOfWeek = now.getDay();
   387→      const mondayOffset = dayOfWeek === 0 ? -6 : 1 - dayOfWeek;
   388→      const monday = new Date(now);
   389→      monday.setDate(now.getDate() + mondayOffset);
   390→      monday.setHours(0, 0, 0, 0);
   391→      const sunday = new Date(monday);
   392→      sunday.setDate(monday.getDate() + 6);
   393→      sunday.setHours(23, 59, 59, 999);
   394→      const evtDate = new Date(evt.date);
   395→
   396→      // Include events from the current week OR events without a specific date match
   397→      // (RSS dates can be approximate, so include anything within a generous range)
   398→      if (evtDate >= monday && evtDate <= sunday) return true;
   399→      // Also include recent events (within last 2 days and next 7 days) for RSS week view
   400→      const twoDaysAgo = new Date(now);
   401→      twoDaysAgo.setDate(now.getDate() - 2);
   402→      const nextWeek = new Date(now);
   403→      nextWeek.setDate(now.getDate() + 7);
   404→      return evtDate >= twoDaysAgo && evtDate <= nextWeek;
   405→    });
   406→
   407→  // Deduplicate by title
   408→  const seen = new Set<string>();
   409→  const uniqueEvents = events.filter((evt: any) => {
   410→    const key = evt.event.toLowerCase().substring(0, 50);
   411→    if (seen.has(key)) return false;
   412→    seen.add(key);
   413→    return true;
   414→  });
   415→
   416→  const highImpactEvents = uniqueEvents.filter((e: any) => e.impact === 'high');
   417→  const nextHighImpact = highImpactEvents.find((e: any) => new Date(e.date) > now) || null;

   680→async function fetchCalendarData(lang: string, period: string = 'today', asset?: string): Promise<{
   681→  events: any[];
   682→  assetEvents?: any[];
   683→  highImpactCount: number;
   684→  nextHighImpact: any | null;
   685→  updatedAt: string;
   686→  error?: string;
   687→}> {
   688→  const isFr = lang === 'fr';
   689→
   690→  // 1. Try ZAI SDK first
   691→  const zaiResult = await fetchCalendarWithZAI(lang, period);
   692→  if (zaiResult && zaiResult.events.length > 0) {
   693→    const result: any = { ...zaiResult };
   694→    if (asset) {
   695→      result.assetEvents = filterEventsForAsset(zaiResult.events, asset);
   696→    }
   697→    return result;
   698→  }
   699→
   700→  // 2. Fallback: RSS feeds from investing.com
   701→  console.log('Calendar: ZAI SDK unavailable or no results, falling back to RSS');
   702→  try {
   703→    const rssItems = await fetchCalendarRSS();
   704→    if (rssItems.length > 0) {
   705→      const result: any = parseCalendarFromRSS(rssItems, lang, period);
   706→      if (asset) {
   707→        result.assetEvents = filterEventsForAsset(result.events, asset);
   708→      }
   709→      return result;
   710→    }
   711→  } catch (error) {
   712→    console.error('Calendar RSS fallback error:', error instanceof Error ? error.message : 'Unknown error');
   713→  }
   714→
   715→  // 3. Both failed
   716→  return {
   717→    events: [],
   718→    assetEvents: asset ? [] : undefined,
   719→    highImpactCount: 0,
   720→    nextHighImpact: null,
   721→    updatedAt: new Date().toISOString(),
   722→    error: isFr ? 'Aucune donnée disponible' : 'No data available',

   724→}
