   189→// ─── Alert Hook ──────────────────────────────────────────
   190→function useAlerts() {
   191→  const [alerts, setAlerts] = useState<any[]>([]);
   192→  const [loading, setLoading] = useState(true);
   193→
   194→  const fetchAlerts = useCallback(async () => {
   195→    setLoading(true);
   196→    try {
   197→      const res = await fetch("/api/alerts");
   198→      if (res.ok) {
   199→        const data = await res.json();
   200→        setAlerts(data.alerts || []);
   201→      }
   202→    } catch {
   203→      setAlerts([]);
   204→    }
   205→    setLoading(false);
   206→  }, []);
   207→
   208→  return { alerts, loading, refetch: fetchAlerts };
   209→}

   211→// ─── Notification Checker ────────────────────────────────
   212→function useNotificationChecker(alerts: any[]) {
   213→  const intervalRef = useRef<NodeJS.Timeout | null>(null);
   214→  const triggeredRef = useRef<Set<string>>(new Set());
   215→
   216→  useEffect(() => {
   217→    if (!alerts.length) return;
   218→
   219→    const checkAlerts = () => {
   220→      const now = new Date();
   221→      alerts.forEach((alert) => {
   222→        if (alert.triggered) return;
   223→        if (triggeredRef.current.has(alert.id)) return;
   224→
   225→        const alertTime = new Date(alert.alertDate);
   226→        if (now >= alertTime) {
   227→          triggeredRef.current.add(alert.id);
   228→          fetch(`/api/alerts/${alert.id}`, {
   229→            method: "PUT",
   230→            headers: { "Content-Type": "application/json" },
   231→            body: JSON.stringify({ triggered: true }),
   232→          }).catch(() => {});
   233→
   234→          if ("Notification" in window) {
   235→            if (Notification.permission === "granted") {
   236→              new Notification("DONCIEL TM — Alerte", {
   237→                body: alert.title + (alert.description ? `\n${alert.description}` : ""),
   238→                icon: "/favicon.ico",
   239→                tag: alert.id,
   240→              });
   241→            } else if (Notification.permission !== "denied") {
   242→              Notification.requestPermission().then((perm) => {
   243→                if (perm === "granted") {
   244→                  new Notification("DONCIEL TM — Alerte", {
   245→                    body: alert.title + (alert.description ? `\n${alert.description}` : ""),
   246→                    icon: "/favicon.ico",
   247→                    tag: alert.id,
   248→                  });
   249→                }
   250→              });
   251→            }
   252→          }
   253→        }
   254→      });
   255→    };
   256→
   257→    checkAlerts();
   258→    intervalRef.current = setInterval(checkAlerts, 15000);
   259→
   260→    return () => {
   261→      if (intervalRef.current) clearInterval(intervalRef.current);
   262→    };
   263→  }, [alerts]);
   264→}

   266→// ─── Performance Hook ────────────────────────────────────
   267→function useNotePerformance() {
   268→  const [data, setData] = useState<any>(null);
   269→  const [loading, setLoading] = useState(false);
   270→
   271→  const fetchPerf = useCallback(async (dateFrom?: string, dateTo?: string) => {
   272→    setLoading(true);
   273→    try {
   274→      const params = new URLSearchParams();
   275→      if (dateFrom) params.set("dateFrom", dateFrom);
   276→      if (dateTo) params.set("dateTo", dateTo);
   277→      const res = await fetch(`/api/notes/performance?${params}`);
   278→      if (res.ok) {
   279→        const result = await res.json();
   280→        setData(result);
   281→      }
   282→    } catch {
   283→      setData(null);
   284→    }
   285→    setLoading(false);
   286→  }, []);
   287→
   288→  return { performanceData: data, loadingPerf: loading, fetchPerf };
   289→}

   291→// ─── Parse helpers ──────────────────────────────────────
   292→function parseJSON<T>(val: string | null | undefined, fallback: T): T {
   293→  if (!val) return fallback;
   294→  try { return JSON.parse(val); } catch { return fallback; }
   295→}

   297→function getNoteGroupLabel(date: Date, language: "fr" | "en"): string {
   298→  if (isToday(date)) return language === "fr" ? "Aujourd'hui" : "Today";
   299→  if (isYesterday(date)) return language === "fr" ? "Hier" : "Yesterday";
   300→  if (isThisWeek(date)) return language === "fr" ? "Cette semaine" : "This week";
   301→  if (isThisMonth(date)) return language === "fr" ? "Ce mois" : "This month";
   302→  return format(date, "MMMM yyyy", { locale: undefined });
   303→}

   329→  const filteredNotes = useMemo(() => {
   330→    let result = notes.filter((n: any) => n.type === activeType);
   331→
   332→    // Search
   333→    if (searchQuery) {
   334→      const q = searchQuery.toLowerCase();
   335→      result = result.filter((n: any) =>
   336→        n.title?.toLowerCase().includes(q) ||
   337→        n.content?.toLowerCase().includes(q) ||
   338→        n.plan?.toLowerCase().includes(q) ||
   339→        n.observation?.toLowerCase().includes(q) ||
   340→        n.rules?.toLowerCase().includes(q) ||
   341→        parseJSON<string[]>(n.tags, []).some(tag => tag.toLowerCase().includes(q))
   342→      );
   343→    }
   344→
   345→    // Bias filter
   346→    if (filterBias) {
   347→      result = result.filter((n: any) => n.marketBias === filterBias);
   348→    }
   349→
   350→    // Priority filter
   351→    if (filterPriority) {
   352→      result = result.filter((n: any) => n.priority === filterPriority);
   353→    }
   354→
   355→    // Sort
   356→    result.sort((a: any, b: any) => {
   357→      // Pinned always first
   358→      if (a.pinned && !b.pinned) return -1;
   359→      if (!a.pinned && b.pinned) return 1;
   360→
   361→      if (sortBy === "priority") {
   362→        const pOrder: Record<string, number> = { URGENT: 4, HIGH: 3, MEDIUM: 2, LOW: 1 };
   363→        return (pOrder[b.priority] || 0) - (pOrder[a.priority] || 0);
   364→      }
   365→      if (sortBy === "mood") {
   366→        return (b.confidence || 0) - (a.confidence || 0);
   367→      }
   368→      return new Date(b.date).getTime() - new Date(a.date).getTime();

   929→// ─── Note Viewer ────────────────────────────────────────
   930→function NoteViewer({
   931→  note, language, onClose, onEdit,
   932→}: {
   933→  note: any;
   934→  language: "fr" | "en";
   935→  onClose: () => void;
   936→  onEdit: () => void;
   937→}) {
   938→  const tags = parseJSON<string[]>(note.tags, []);
   939→  const checklist = parseJSON<{ text: string; checked: boolean }[]>(note.checklist, []);
   940→  const biasDef = BIASES.find(b => b.id === note.marketBias);
   941→  const priorityDef = PRIORITIES.find(p => p.id === note.priority);
   942→  const moodDef = MOODS.find(m => m.emoji === note.mood);
   943→
   944→  const [checklistState, setChecklistState] = useState(checklist);
   945→  const [savingChecklist, setSavingChecklist] = useState(false);
   946→
   947→  const handleToggleCheck = async (index: number) => {
   948→    const updated = [...checklistState];
   949→    updated[index] = { ...updated[index], checked: !updated[index].checked };
   950→    setChecklistState(updated);
   951→    setSavingChecklist(true);
   952→    try {
   953→      await fetch(`/api/notes/${note.id}`, {
   954→        method: "PUT",
   955→        headers: { "Content-Type": "application/json" },
   956→        body: JSON.stringify({ checklist: updated }),
   957→      });
   958→    } catch {
   959→      // Revert on error
   960→      setChecklistState(checklist);
   961→    }
   962→    setSavingChecklist(false);
   963→  };
   964→
   965→  const checkedCount = checklistState.filter(c => c.checked).length;
   966→  const progress = checklistState.length > 0 ? (checkedCount / checklistState.length) * 100 : 0;
   967→
   968→  return (
   969→    <>
   970→      <DialogHeader>
   971→        <DialogTitle className="flex items-center gap-2 flex-wrap">
   972→          {note.pinned && <Pin className="w-4 h-4 text-amber-500" />}
   973→          <span className="text-xl">{note.mood || "📝"}</span>
   974→          {note.title}
   975→          {biasDef && (
   976→            <Badge variant="outline" className={cn("text-[10px] gap-1", biasDef.color)}>
   977→              {React.createElement(biasDef.icon, { className: "w-3 h-3" })}
   978→              {language === "fr" ? biasDef.label_fr : biasDef.label_en}
   979→            </Badge>
   980→          )}
   981→          {priorityDef && (
   982→            <Badge variant="outline" className={cn("text-[10px]", priorityDef.color)}>
