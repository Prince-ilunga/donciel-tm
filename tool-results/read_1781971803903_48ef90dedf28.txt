   189→// ─── Alert Hook ──────────────────────────────────────────
   190→function useAlerts() {
   191→  const [alerts, setAlerts] = useState<any[]>([]);
   192→  const [loading, setLoading] = useState(true);
   193→
   194→  const fetchAlerts = useCallback(async () => {
   195→    setLoading(true);
   196→    try {
   197→      const res = await fetch("/api/alerts");
   198→      if (res.ok) {
   199→        const data = await res.json();
   200→        setAlerts(data.alerts || []);
   201→      }
   202→    } catch {
   203→      setAlerts([]);
   204→    }
   205→    setLoading(false);
   206→  }, []);
   207→
   208→  return { alerts, loading, refetch: fetchAlerts };
   209→}

   211→// ─── Notification Checker ────────────────────────────────
   212→function useNotificationChecker(alerts: any[]) {
   213→  const intervalRef = useRef<NodeJS.Timeout | null>(null);
   214→  const triggeredRef = useRef<Set<string>>(new Set());
   215→
   216→  useEffect(() => {
   217→    if (!alerts.length) return;
   218→
   219→    const checkAlerts = () => {
   220→      const now = new Date();
   221→      alerts.forEach((alert) => {
   222→        if (alert.triggered) return;
   223→        if (triggeredRef.current.has(alert.id)) return;
   224→
   225→        const alertTime = new Date(alert.alertDate);
   226→        if (now >= alertTime) {
   227→          triggeredRef.current.add(alert.id);
   228→          fetch(`/api/alerts/${alert.id}`, {
   229→            method: "PUT",
   230→            headers: { "Content-Type": "application/json" },
   231→            body: JSON.stringify({ triggered: true }),
   232→          }).catch(() => {});
   233→
   234→          if ("Notification" in window) {
   235→            if (Notification.permission === "granted") {
   236→              new Notification("DONCIEL TM — Alerte", {
   237→                body: alert.title + (alert.description ? `\n${alert.description}` : ""),
   238→                icon: "/favicon.ico",
   239→                tag: alert.id,
   240→              });
   241→            } else if (Notification.permission !== "denied") {
   242→              Notification.requestPermission().then((perm) => {
   243→                if (perm === "granted") {
   244→                  new Notification("DONCIEL TM — Alerte", {
   245→                    body: alert.title + (alert.description ? `\n${alert.description}` : ""),
   246→                    icon: "/favicon.ico",
   247→                    tag: alert.id,
   248→                  });
   249→                }
   250→              });
   251→            }
   252→          }
   253→        }
   254→      });
   255→    };
   256→
   257→    checkAlerts();
   258→    intervalRef.current = setInterval(checkAlerts, 15000);
   259→
   260→    return () => {
   261→      if (intervalRef.current) clearInterval(intervalRef.current);
   262→    };
   263→  }, [alerts]);
   264→}

   267→function useNotePerformance() {
   268→  const [data, setData] = useState<any>(null);
   269→  const [loading, setLoading] = useState(false);
   270→
   271→  const fetchPerf = useCallback(async (dateFrom?: string, dateTo?: string) => {

   275→      if (dateFrom) params.set("dateFrom", dateFrom);
   276→      if (dateTo) params.set("dateTo", dateTo);
   277→      const res = await fetch(`/api/notes/performance?${params}`);
   278→      if (res.ok) {
   279→        const result = await res.json();
   280→        setData(result);
   281→      }
   282→    } catch {
   283→      setData(null);
   284→    }
   285→    setLoading(false);
   286→  }, []);
   287→
   288→  return { performanceData: data, loadingPerf: loading, fetchPerf };

   292→function parseJSON<T>(val: string | null | undefined, fallback: T): T {
   293→  if (!val) return fallback;
   294→  try { return JSON.parse(val); } catch { return fallback; }

   297→function getNoteGroupLabel(date: Date, language: "fr" | "en"): string {
   298→  if (isToday(date)) return language === "fr" ? "Aujourd'hui" : "Today";
   299→  if (isYesterday(date)) return language === "fr" ? "Hier" : "Yesterday";
   300→  if (isThisWeek(date)) return language === "fr" ? "Cette semaine" : "This week";
   301→  if (isThisMonth(date)) return language === "fr" ? "Ce mois" : "This month";
   302→  return format(date, "MMMM yyyy", { locale: undefined });
   303→}

   305→// ─── Error Boundary ─────────────────────────────────────
   306→interface ErrorBoundaryState {
   307→  hasError: boolean;
   308→  error: Error | null;
   309→}
   310→
   311→class NotesErrorBoundary extends React.Component<
   312→  { children: React.ReactNode; language: "fr" | "en" },
   313→  ErrorBoundaryState
   314→> {
   315→  constructor(props: { children: React.ReactNode; language: "fr" | "en" }) {
   316→    super(props);
   317→    this.state = { hasError: false, error: null };
   318→  }
   319→
   320→  static getDerivedStateFromError(error: Error): ErrorBoundaryState {
   321→    return { hasError: true, error };
   322→  }
   323→
   324→  componentDidCatch(error: Error, errorInfo: React.ErrorInfo) {
   325→    console.error("NotesTab error:", error, errorInfo);
   326→  }
   327→
   328→  handleRetry = () => {
   329→    this.setState({ hasError: false, error: null });
   330→  };
   331→
   332→  render() {
   333→    if (this.state.hasError) {
   334→      const { language } = this.props;
   335→      return (
   336→        <div className="p-6 max-w-md mx-auto text-center space-y-4">
   337→          <AlertTriangle className="w-12 h-12 mx-auto text-amber-500" />
   338→          <h3 className="text-lg font-semibold">
   339→            {language === "fr" ? "Une erreur est survenue" : "An error occurred"}
   340→          </h3>
   341→          <p className="text-sm text-muted-foreground">
   342→            {language === "fr"
   343→              ? "Le chargement des notes a rencontré un problème. Vous pouvez réessayer."
   344→              : "Loading notes encountered a problem. You can retry."}
   345→          </p>
   346→          {this.state.error && (
   347→            <p className="text-xs text-muted-foreground bg-muted/50 p-2 rounded font-mono break-all">
   348→              {this.state.error.message}
   349→            </p>
   350→          )}
   351→          <div className="flex gap-2 justify-center">
   352→            <Button onClick={this.handleRetry} variant="default" size="sm">
   353→              {language === "fr" ? "Réessayer" : "Retry"}
   354→            </Button>
   355→            <Button onClick={() => window.location.reload()} variant="outline" size="sm">
   356→              {language === "fr" ? "Recharger la page" : "Reload page"}
   357→            </Button>
   358→          </div>
   359→        </div>
   360→      );
   361→    }
   362→    return this.props.children;
   363→  }
   364→}

   389→  // Filtered and sorted notes
   390→  const filteredNotes = useMemo(() => {
   391→    let result = notes.filter((n: any) => n.type === activeType);
   392→
   393→    // Search
   394→    if (searchQuery) {
   395→      const q = searchQuery.toLowerCase();
   396→      result = result.filter((n: any) =>
   397→        n.title?.toLowerCase().includes(q) ||
   398→        n.content?.toLowerCase().includes(q) ||
   399→        n.plan?.toLowerCase().includes(q) ||
   400→        n.observation?.toLowerCase().includes(q) ||
   401→        n.rules?.toLowerCase().includes(q) ||
   402→        parseJSON<string[]>(n.tags, []).some(tag => tag.toLowerCase().includes(q))
   403→      );
   404→    }
   405→
   406→    // Bias filter
   407→    if (filterBias) {
   408→      result = result.filter((n: any) => n.marketBias === filterBias);
   409→    }
   410→
   411→    // Priority filter
   412→    if (filterPriority) {
   413→      result = result.filter((n: any) => n.priority === filterPriority);
   414→    }
   415→
   416→    // Sort
   417→    result.sort((a: any, b: any) => {
   418→      // Pinned always first
   419→      if (a.pinned && !b.pinned) return -1;
   420→      if (!a.pinned && b.pinned) return 1;
   421→
   422→      if (sortBy === "priority") {
   423→        const pOrder: Record<string, number> = { URGENT: 4, HIGH: 3, MEDIUM: 2, LOW: 1 };
   424→        return (pOrder[b.priority] || 0) - (pOrder[a.priority] || 0);
   425→      }
   426→      if (sortBy === "mood") {
   427→        return (b.confidence || 0) - (a.confidence || 0);
   428→      }
   429→      return new Date(b.date).getTime() - new Date(a.date).getTime();
   430→    });
   431→
   432→    return result;

   881→// ─── Note Card ──────────────────────────────────────────
   882→function NoteCard({
   883→  note, language, onView, onEdit, onDelete, onTogglePin,
   884→}: {
   885→  note: any;
   886→  language: "fr" | "en";
   887→  onView: () => void;
   888→  onEdit: () => void;
   889→  onDelete: () => void;
   890→  onTogglePin: () => void;
   891→}) {
   892→  const tags = parseJSON<string[]>(note.tags, []);
   893→  const checklist = parseJSON<{ text: string; checked: boolean }[]>(note.checklist, []);
   894→  const checkedCount = checklist.filter(c => c.checked).length;
   895→  const priorityDef = PRIORITIES.find(p => p.id === note.priority);
   896→  const biasDef = BIASES.find(b => b.id === note.marketBias);
   897→
   898→  return (
   899→    <Card
   900→      className={cn(
   901→        "p-3 hover:border-primary/30 transition-all cursor-pointer group",
   902→        note.pinned && "border-amber-500/30 bg-amber-500/5",
   903→        note.priority === "URGENT" && "border-red-500/30"
   904→      )}
   905→      onClick={onView}
   906→    >
   907→      <div className="flex items-start gap-3">
   908→        {/* Mood indicator */}
   909→        <div className="shrink-0 text-xl w-9 h-9 flex items-center justify-center rounded-lg bg-muted/50">
   910→          {note.mood || "📝"}
   911→        </div>
   912→
   913→        {/* Content */}
   914→        <div className="flex-1 min-w-0">
   915→          <div className="flex items-center gap-1.5 flex-wrap mb-1">
   916→            {note.pinned && <Pin className="w-3 h-3 text-amber-500 shrink-0" />}
   917→            <h4 className="text-sm font-semibold truncate">{note.title}</h4>
   918→            {biasDef && (
   919→              <Badge variant="outline" className={cn("text-[9px] gap-0.5 px-1.5 py-0", biasDef.color)}>
   920→                {React.createElement(biasDef.icon, { className: "w-2.5 h-2.5" })}
   921→              </Badge>
   922→            )}
   923→            {priorityDef && (
   924→              <Badge variant="outline" className={cn("text-[9px] px-1.5 py-0", priorityDef.color)}>
   925→                {language === "fr" ? priorityDef.label_fr : priorityDef.label_en}
   926→              </Badge>
   927→            )}
   928→          </div>
   929→
   930→          {/* Preview text */}
   931→          <p className="text-xs text-muted-foreground line-clamp-2 mb-1.5 whitespace-pre-wrap">
   932→            {note.content || note.plan || note.observation || ""}
   933→          </p>
   934→
   935→          {/* Meta row */}
   936→          <div className="flex items-center gap-2 flex-wrap">
   937→            <Badge variant="outline" className="text-[9px]">
   938→              {new Date(note.date).toLocaleDateString(language === "fr" ? "fr-FR" : "en-US", { month: "short", day: "numeric" })}
   939→            </Badge>
   940→            {note.confidence != null && (
   941→              <Badge variant="secondary" className="text-[9px] gap-0.5">
   942→                <Target className="w-2.5 h-2.5" />
   943→                {note.confidence}/10
   944→              </Badge>
   945→            )}
   946→            {checklist.length > 0 && (
   947→              <Badge variant="secondary" className="text-[9px] gap-0.5">
   948→                <CheckSquare className="w-2.5 h-2.5" />
   949→                {checkedCount}/{checklist.length}
   950→              </Badge>
   951→            )}
   952→            {note.screenshots?.length > 0 && (
   953→              <Badge variant="secondary" className="text-[9px] gap-0.5">
   954→                <Camera className="w-2.5 h-2.5" />
   955→                {note.screenshots.length}
   956→              </Badge>
   957→            )}
   958→            {note.alerts?.filter((a: any) => !a.triggered).length > 0 && (
   959→              <Badge variant="secondary" className="text-[9px] gap-0.5">
   960→                <Bell className="w-2.5 h-2.5" />
   961→                {note.alerts.filter((a: any) => !a.triggered).length}
   962→              </Badge>
   963→            )}
   964→            {tags.slice(0, 3).map((tag: string) => (
   965→              <Badge key={tag} variant="outline" className="text-[9px] px-1.5 py-0 text-primary border-primary/20">
   966→                {tag}
   967→              </Badge>
   968→            ))}
   969→            {tags.length > 3 && (
   970→              <span className="text-[9px] text-muted-foreground">+{tags.length - 3}</span>
   971→            )}
   972→          </div>
   973→        </div>
   974→
   975→        {/* Actions */}
   976→        <div className="flex items-center gap-0.5 shrink-0 opacity-0 group-hover:opacity-100 transition-opacity">
   977→          <Button variant="ghost" size="icon" className="h-7 w-7" onClick={(e) => { e.stopPropagation(); onTogglePin(); }}>
   978→            {note.pinned ? <PinOff className="w-3 h-3 text-amber-500" /> : <Pin className="w-3 h-3" />}
   979→          </Button>
   980→          <Button variant="ghost" size="icon" className="h-7 w-7" onClick={(e) => { e.stopPropagation(); onEdit(); }}>
   981→            <Edit3 className="w-3 h-3" />
   982→          </Button>
   983→          <Button variant="ghost" size="icon" className="h-7 w-7 text-loss" onClick={(e) => { e.stopPropagation(); onDelete(); }}>
   984→            <Trash2 className="w-3 h-3" />
   985→          </Button>
   986→        </div>
   987→      </div>
