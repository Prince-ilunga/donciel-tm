   125→function generateAssetBriefing(
   126→  asset: string,
   127→  sentiment: 'bullish' | 'bearish' | 'neutral',
   128→  lang: string,
   129→  rssItems: { title: string; snippet: string; date: string }[]
   130→): { keyLevels: string[]; scenarioAdjustment: string; riskEvents: string[] } | null {
   131→  const isFr = lang === 'fr';
   132→  const data = ASSET_BRIEFING_DATA[asset.toUpperCase()];
   133→  if (!data) return null;
   134→
   135→  const keyLevels = isFr ? data.keyLevelsFr : data.keyLevelsEn;
   136→
   137→  const scenarioAdjustment = sentiment === 'bullish'
   138→    ? (isFr ? data.bullAdjFr : data.bullAdjEn)
   139→    : sentiment === 'bearish'
   140→    ? (isFr ? data.bearAdjFr : data.bearAdjEn)
   141→    : (isFr ? data.neutralAdjFr : data.neutralAdjEn);
   142→
   143→  // Filter risk events relevant to this asset
   144→  const riskEvents = rssItems
   145→    .filter(i => {
   146→      const t = i.title.toLowerCase();
   147→      return data.riskKeywords.some(kw => t.includes(kw));
   148→    })
   149→    .slice(0, 5)
   150→    .map(i => i.title.length > 80 ? i.title.substring(0, 80) + '...' : i.title);
   151→
   152→  if (riskEvents.length === 0) {
   153→    riskEvents.push(
   154→      isFr
   155→        ? `Surveillez les annonces impactant ${asset.toUpperCase()}`
   156→        : `Watch for announcements impacting ${asset.toUpperCase()}`,
   157→      isFr
   158→        ? 'Calendrier économique à suivre'
   159→        : 'Economic calendar to watch'
   160→    );
   161→  }
   162→
   163→  return { keyLevels, scenarioAdjustment, riskEvents };
   164→}

   306→async function fetchBriefingRSS(): Promise<{ title: string; snippet: string; date: string }[]> {
   307→  const items: { title: string; snippet: string; date: string }[] = [];
   308→
   309→  const feedResults = await Promise.allSettled(
   310→    BRIEFING_RSS_FEEDS.map(feed =>
   311→      fetch(`https://www.investing.com/rss/news_${feed.id}.rss`, {
   312→        headers: { 'User-Agent': 'Mozilla/5.0 (compatible; DONCIEL-TM/1.0)' },
   313→        signal: AbortSignal.timeout(10000),
   314→      }).then(res => {
   315→        if (!res.ok) return [];
   316→        return res.text().then(xml => {
   317→          const parsed = parser.parse(xml);
   318→          const feedItems = parsed?.rss?.channel?.item;
   319→          const list = Array.isArray(feedItems) ? feedItems : feedItems ? [feedItems] : [];
   320→          return list
   321→            .filter((item: any) => {
   322→              const title = (item.title || '').toLowerCase();
   323→              return feed.keywords.some(kw => title.includes(kw));
   324→            })
   325→            .map((item: any) => ({
   326→              title: item.title || '',
   327→              snippet: item.title || '',
   328→              date: item.pubDate || '',
   329→            }));
   330→        });
   331→      }).catch(() => [])
   332→    )
   333→  );
   334→
   335→  for (const result of feedResults) {
   336→    if (result.status === 'fulfilled' && Array.isArray(result.value)) {
   337→      items.push(...result.value);
   338→    }
   339→  }
   340→
   341→  // Deduplicate by title
   342→  const seen = new Set<string>();
   343→  return items.filter(item => {
   344→    const key = item.title.toLowerCase().substring(0, 50);
   345→    if (seen.has(key)) return false;
   346→    seen.add(key);
   347→    return true;
   348→  });
   349→}

   441→    const extractData = (result: PromiseSettledResult<any>) => {
   442→      if (result.status === 'fulfilled' && result.value && Array.isArray(result.value)) {
   443→        return result.value
   444→          .map((r: any) => `${r.title || ''}: ${r.snippet || r.description || ''}`)
   445→          .join('\n');
   446→      }
   447→      return '';
   448→    };
   449→
   450→    const overnightData = extractData(overnightResult);
   451→    const outlookData = extractData(outlookResult);
   452→    const technicalData = extractData(technicalResult);
   453→
   454→    const allData = [overnightData, outlookData, technicalData].filter(Boolean);
   455→    if (allData.length === 0) {
   456→      return null; // Trigger fallback
   457→    }
   458→
   459→    const systemPrompt = isFr ? BRIEFING_SYSTEM_PROMPT_FR : BRIEFING_SYSTEM_PROMPT_EN;
   460→    const combinedData = [
   461→      overnightData ? `=== SESSION ASIATIQUE / OVERNIGHT ===\n${overnightData}` : '',

   463→      technicalData ? `=== NIVEAUX CLÉS / TECHNIQUE ===\n${technicalData}` : '',

   466→    const todayDate = new Date();

   468→    const dayNamesEn = ['Sunday', 'Monday', 'Tuesday', 'Wednesday', 'Thursday', 'Friday', 'Saturday'];
   469→    const dayName = isFr ? dayNames[todayDate.getDay()] : dayNamesEn[todayDate.getDay()];
   470→
   471→    const completion = await zai.chat.completions.create({

   473→        { role: 'assistant', content: systemPrompt },
   474→        {
   475→          role: 'user',
   476→          content: isFr

   479→DONNÉES COLLECTÉES:

   483→{
   484→  "summary": "Résumé en 3-4 phrases avec données chiffrées",
   485→  "asia": "Résumé session asiatique en 2-3 phrases",
   486→  "today": "Événements clés attendus en 2-3 phrases",
   487→  "keyLevels": ["EUR/USD: Support / Résistance", "Or: Support / Résistance", "S&P 500: Support / Résistance"],

   538→    return null; // Trigger fallback

   546→async function fetchBriefingData(lang: string, period: string = 'today', asset?: string): Promise<{
   547→  summary: string;
   548→  asia: string;
   549→  today: string;
   550→  keyLevels: string[];
   551→  scenarios: { name: string; probability: number; description: string }[];
   552→  riskEvents: string[];
   553→  assetBriefing?: { keyLevels: string[]; scenarioAdjustment: string; riskEvents: string[] };
   554→  updatedAt: string;
   555→  error?: string;
   556→}> {
   557→  const isFr = lang === 'fr';
   558→
   559→  const fallbackResult = {
   560→    summary: isFr ? 'Données de briefing temporairement indisponibles.' : 'Briefing data temporarily unavailable.',
   561→    asia: isFr ? 'Session asiatique : données non disponibles.' : 'Asian session: data unavailable.',
   562→    today: isFr ? 'Événements du jour : données non disponibles.' : "Today's events: data unavailable.",
   563→    keyLevels: [],
   564→    scenarios: [
   565→      { name: isFr ? 'Haussier' : 'Bullish', probability: 33, description: isFr ? 'Données insuffisantes' : 'Insufficient data' },
   566→      { name: 'Neutre', probability: 34, description: isFr ? 'Données insuffisantes' : 'Insufficient data' },
   567→      { name: isFr ? 'Baissier' : 'Bearish', probability: 33, description: isFr ? 'Données insuffisantes' : 'Insufficient data' },
   568→    ],
   569→    riskEvents: [],
   570→    updatedAt: new Date().toISOString(),
   571→  };
   572→
   573→  // Helper to determine sentiment for asset briefing
   574→  const getSentiment = (data: any): 'bullish' | 'bearish' | 'neutral' => {
   575→    if (data.overallSentiment === 'RISK-ON') return 'bullish';
   576→    if (data.overallSentiment === 'RISK-OFF') return 'bearish';
   577→    return 'neutral';
   578→  };
   579→
   580→  // 1. Try ZAI SDK first
   581→  const zaiResult = await fetchBriefingWithZAI(lang, period);
   582→  if (zaiResult) {
   583→    const result: any = { ...zaiResult };
   584→    // Add asset-specific briefing if requested
   585→    if (asset) {
   586→      try {
   587→        const rssItems = await fetchBriefingRSS();
   588→        const sentiment = getSentiment(zaiResult);
   589→        result.assetBriefing = generateAssetBriefing(asset, sentiment, lang, rssItems);
   590→      } catch {}
   591→    }
   592→    return result;
   593→  }
   594→
   595→  // 2. Fallback: RSS feeds + keyword analysis
   596→  console.log('Briefing: ZAI SDK unavailable, falling back to RSS');
   597→  try {
   598→    const rssItems = await fetchBriefingRSS();
   599→    if (rssItems.length > 0) {
   600→      const result: any = generateBriefingFromRSS(rssItems, lang, period);
   601→      // Add asset-specific briefing if requested
   602→      if (asset) {
   603→        const text = rssItems.map(i => i.title.toLowerCase()).join(' ');
   604→        const bullWords = ['rally', 'surge', 'gain', 'rise', 'bullish', 'recovery', 'optimism', 'boost'];
   605→        const bearWords = ['crash', 'selloff', 'fear', 'decline', 'bearish', 'recession', 'slump', 'plunge'];
   606→        let bs = 0;
   607→        let brs = 0;
   608→        bullWords.forEach(w => { const m = text.match(new RegExp(w, 'gi')); if (m) bs += m.length; });
   609→        bearWords.forEach(w => { const m = text.match(new RegExp(w, 'gi')); if (m) brs += m.length; });
   610→        const sentiment: 'bullish' | 'bearish' | 'neutral' = bs > brs * 1.3 ? 'bullish' : brs > bs * 1.3 ? 'bearish' : 'neutral';
   611→        result.assetBriefing = generateAssetBriefing(asset, sentiment, lang, rssItems);
   612→      }
   613→      return result;
   614→    }
   615→  } catch (error) {
   616→    console.error('Briefing RSS fallback error:', error instanceof Error ? error.message : 'Unknown error');
   617→  }
   618→
   619→  // 3. Both failed
   620→  const failResult: any = { ...fallbackResult, error: isFr ? 'Aucune donnée de marché disponible' : 'No market data available' };
   621→  if (asset) {
   622→    failResult.assetBriefing = generateAssetBriefing(asset, 'neutral', lang, []);
   623→  }
   624→  return failResult;
   625→}

   631→      return NextResponse.json({ error: result.error }, { status: 401 });

   633→

   635→    const lang = searchParams.get('lang') === 'en' ? 'en' : 'fr';

   637→    const asset = searchParams.get('asset') || undefined;
   638→
   639→    // Check cache

   641→    const cached = cache.get(cacheKey);
   642→    const cacheDuration = period === 'week' ? CACHE_DURATION_WEEK : CACHE_DURATION;
   643→    if (cached && Date.now() - cached.timestamp < cacheDuration) {

   648→
   649→    cache.set(cacheKey, { data, timestamp: Date.now() });
