    49→function generateAssetBriefing(
    50→  asset: string,
    51→  sentiment: 'bullish' | 'bearish' | 'neutral',
    52→  lang: string,
    53→  rssItems: { title: string; snippet: string; date: string }[]
    54→): { keyLevels: string[]; scenarioAdjustment: string; riskEvents: string[] } | null {
    55→  const isFr = lang === 'fr';
    56→  const data = ASSET_BRIEFING_DATA[asset.toUpperCase()];
    57→  if (!data) return null;
    58→
    59→  const keyLevels = isFr ? data.keyLevelsFr : data.keyLevelsEn;
    60→
    61→  const scenarioAdjustment = sentiment === 'bullish'
    62→    ? (isFr ? data.bullAdjFr : data.bullAdjEn)
    63→    : sentiment === 'bearish'
    64→    ? (isFr ? data.bearAdjFr : data.bearAdjEn)
    65→    : (isFr ? data.neutralAdjFr : data.neutralAdjEn);
    66→
    67→  // Filter risk events relevant to this asset
    68→  const riskEvents = rssItems
    69→    .filter(i => {
    70→      const t = i.title.toLowerCase();
    71→      return data.riskKeywords.some(kw => t.includes(kw));
    72→    })
    73→    .slice(0, 5)
    74→    .map(i => i.title.length > 80 ? i.title.substring(0, 80) + '...' : i.title);
    75→
    76→  if (riskEvents.length === 0) {
    77→    riskEvents.push(
    78→      isFr
    79→        ? `Surveillez les annonces impactant ${asset.toUpperCase()}`
    80→        : `Watch for announcements impacting ${asset.toUpperCase()}`,
    81→      isFr
    82→        ? 'Calendrier économique à suivre'
    83→        : 'Economic calendar to watch'
    84→    );
    85→  }
    86→
    87→  return { keyLevels, scenarioAdjustment, riskEvents };
    88→}

   230→async function fetchBriefingRSS(): Promise<{ title: string; snippet: string; date: string }[]> {
   231→  const items: { title: string; snippet: string; date: string }[] = [];
   232→
   233→  const feedResults = await Promise.allSettled(
   234→    BRIEFING_RSS_FEEDS.map(feed =>
   235→      fetch(`https://www.investing.com/rss/news_${feed.id}.rss`, {
   236→        headers: { 'User-Agent': 'Mozilla/5.0 (compatible; DONCIEL-TM/1.0)' },
   237→        signal: AbortSignal.timeout(10000),
   238→      }).then(res => {
   239→        if (!res.ok) return [];
   240→        return res.text().then(xml => {
   241→          const parsed = parser.parse(xml);
   242→          const feedItems = parsed?.rss?.channel?.item;
   243→          const list = Array.isArray(feedItems) ? feedItems : feedItems ? [feedItems] : [];
   244→          return list
   245→            .filter((item: any) => {
   246→              const title = (item.title || '').toLowerCase();
   247→              return feed.keywords.some(kw => title.includes(kw));
   248→            })
   249→            .map((item: any) => ({
   250→              title: item.title || '',
   251→              snippet: item.title || '',
   252→              date: item.pubDate || '',
   253→            }));
   254→        });
   255→      }).catch(() => [])
   256→    )
   257→  );
   258→
   259→  for (const result of feedResults) {
   260→    if (result.status === 'fulfilled' && Array.isArray(result.value)) {
   261→      items.push(...result.value);
   262→    }
   263→  }
   264→
   265→  // Deduplicate by title
   266→  const seen = new Set<string>();
   267→  return items.filter(item => {
   268→    const key = item.title.toLowerCase().substring(0, 50);
   269→    if (seen.has(key)) return false;
   270→    seen.add(key);
   271→    return true;
   272→  });
   273→}

   452→          scenarios: Array.isArray(parsed.scenarios) ? parsed.scenarios : fallbackResult.scenarios,
   453→          riskEvents: Array.isArray(parsed.riskEvents) ? parsed.riskEvents : [],
   454→          updatedAt: new Date().toISOString(),
   455→        };
   456→      }
   457→    } catch {}
   458→
   459→    return fallbackResult;
   460→  } catch (error) {
   461→    console.error('Briefing ZAI error:', error instanceof Error ? error.message : 'Unknown error');
   462→    return null; // Trigger fallback
   463→  }
   464→}

   470→async function fetchBriefingData(lang: string, period: string = 'today', asset?: string): Promise<{
   471→  summary: string;
   472→  asia: string;
   473→  today: string;
   474→  keyLevels: string[];
   475→  scenarios: { name: string; probability: number; description: string }[];
   476→  riskEvents: string[];
   477→  assetBriefing?: { keyLevels: string[]; scenarioAdjustment: string; riskEvents: string[] };
   478→  updatedAt: string;
   479→  error?: string;
   480→}> {
   481→  const isFr = lang === 'fr';
   482→
   483→  const fallbackResult = {
   484→    summary: isFr ? 'Données de briefing temporairement indisponibles.' : 'Briefing data temporarily unavailable.',
   485→    asia: isFr ? 'Session asiatique : données non disponibles.' : 'Asian session: data unavailable.',
   486→    today: isFr ? 'Événements du jour : données non disponibles.' : "Today's events: data unavailable.",
   487→    keyLevels: [],
   488→    scenarios: [
   489→      { name: isFr ? 'Haussier' : 'Bullish', probability: 33, description: isFr ? 'Données insuffisantes' : 'Insufficient data' },
   490→      { name: 'Neutre', probability: 34, description: isFr ? 'Données insuffisantes' : 'Insufficient data' },
   491→      { name: isFr ? 'Baissier' : 'Bearish', probability: 33, description: isFr ? 'Données insuffisantes' : 'Insufficient data' },
   492→    ],
   493→    riskEvents: [],
   494→    updatedAt: new Date().toISOString(),
   495→  };
   496→
   497→  // Helper to determine sentiment for asset briefing
   498→  const getSentiment = (data: any): 'bullish' | 'bearish' | 'neutral' => {
   499→    if (data.overallSentiment === 'RISK-ON') return 'bullish';
   500→    if (data.overallSentiment === 'RISK-OFF') return 'bearish';
   501→    return 'neutral';
   502→  };
   503→
   504→  // 1. Try ZAI SDK first
   505→  const zaiResult = await fetchBriefingWithZAI(lang, period);
   506→  if (zaiResult) {
   507→    const result: any = { ...zaiResult };
   508→    // Add asset-specific briefing if requested
   509→    if (asset) {
   510→      try {
   511→        const rssItems = await fetchBriefingRSS();
   512→        const sentiment = getSentiment(zaiResult);
   513→        result.assetBriefing = generateAssetBriefing(asset, sentiment, lang, rssItems);
   514→      } catch {}
   515→    }
   516→    return result;
   517→  }
   518→
   519→  // 2. Fallback: RSS feeds + keyword analysis
   520→  console.log('Briefing: ZAI SDK unavailable, falling back to RSS');
   521→  try {
   522→    const rssItems = await fetchBriefingRSS();
   523→    if (rssItems.length > 0) {
   524→      const result: any = generateBriefingFromRSS(rssItems, lang, period);
   525→      // Add asset-specific briefing if requested
   526→      if (asset) {
   527→        const text = rssItems.map(i => i.title.toLowerCase()).join(' ');
   528→        const bullWords = ['rally', 'surge', 'gain', 'rise', 'bullish', 'recovery', 'optimism', 'boost'];
   529→        const bearWords = ['crash', 'selloff', 'fear', 'decline', 'bearish', 'recession', 'slump', 'plunge'];
   530→        let bs = 0;
   531→        let brs = 0;
   532→        bullWords.forEach(w => { const m = text.match(new RegExp(w, 'gi')); if (m) bs += m.length; });
   533→        bearWords.forEach(w => { const m = text.match(new RegExp(w, 'gi')); if (m) brs += m.length; });
   534→        const sentiment: 'bullish' | 'bearish' | 'neutral' = bs > brs * 1.3 ? 'bullish' : brs > bs * 1.3 ? 'bearish' : 'neutral';
   535→        result.assetBriefing = generateAssetBriefing(asset, sentiment, lang, rssItems);
   536→      }
   537→      return result;
   538→    }
   539→  } catch (error) {
   540→    console.error('Briefing RSS fallback error:', error instanceof Error ? error.message : 'Unknown error');
   541→  }
   542→
   543→  // 3. Both failed
   544→  const failResult: any = { ...fallbackResult, error: isFr ? 'Aucune donnée de marché disponible' : 'No market data available' };
   545→  if (asset) {
   546→    failResult.assetBriefing = generateAssetBriefing(asset, 'neutral', lang, []);
   547→  }
   548→  return failResult;
   549→}

   552→  try {
   553→    const result = await getAuthUser();
   554→    if (result.error) {
   555→      return NextResponse.json({ error: result.error }, { status: 401 });
   556→    }
   557→
   558→    const { searchParams } = new URL(request.url);
   559→    const lang = searchParams.get('lang') === 'en' ? 'en' : 'fr';
   560→    const period = searchParams.get('period') === 'week' ? 'week' : 'today';
   561→    const asset = searchParams.get('asset') || undefined;
   562→
   563→    // Check cache
   564→    const cacheKey = `briefing-${lang}-${period}${asset ? `-${asset}` : ''}`;
   565→    const cached = cache.get(cacheKey);
   566→    const cacheDuration = period === 'week' ? CACHE_DURATION_WEEK : CACHE_DURATION;
   567→    if (cached && Date.now() - cached.timestamp < cacheDuration) {
   568→      return NextResponse.json(cached.data);
   569→    }
   570→
   571→    const data = await fetchBriefingData(lang, period, asset);
   572→
   573→    cache.set(cacheKey, { data, timestamp: Date.now() });
   574→    return NextResponse.json(data);
   575→  } catch (error) {
   576→    console.error('Briefing route error:', error);
   577→    return NextResponse.json(
   578→      {
   579→        error: 'Erreur lors de la récupération du briefing',
   580→        summary: 'Briefing temporairement indisponible.',
   581→        asia: 'Données non disponibles.',
   582→        today: 'Données non disponibles.',
   583→        keyLevels: [],
   584→        scenarios: [
   585→          { name: 'Haussier', probability: 33, description: 'Données insuffisantes' },
   586→          { name: 'Neutre', probability: 34, description: 'Données insuffisantes' },
   587→          { name: 'Baissier', probability: 33, description: 'Données insuffisantes' },
   588→        ],
   589→        riskEvents: [],
   590→        updatedAt: new Date().toISOString(),
   591→      },
   592→      { status: 500 }
   593→    );
   594→  }
