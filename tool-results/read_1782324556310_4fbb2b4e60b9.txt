   441→   441→async function fetchSentimentData(lang: string, period: string = 'today', asset?: string): Promise<{
   442→   442→  fearGreed: { value: number; label: string; trend: string };
   443→   443→  vix: { value: number; trend: string; interpretation: string };
   444→   444→  smartMoney: { direction: string; confidence: string };
   445→   445→  retail: { direction: string; confidence: string };
   446→   446→  contrarianSignal: string;
   447→   447→  overallSentiment: string;
   448→   448→  interpretation: string;
   449→   449→  assetSentiment?: { direction: string; confidence: string; interpretation: string };
   450→   450→  updatedAt: string;
   451→   451→  error?: string;
   452→   452→}> {
   453→   453→  const isFr = lang === 'fr';
   454→   454→
   455→   455→  const fallbackResult = {
   456→   456→    fearGreed: { value: 50, label: isFr ? 'Neutre' : 'Neutral', trend: 'stable' },
   457→   457→    vix: { value: 0, trend: 'stable', interpretation: isFr ? 'Données non disponibles' : 'Data unavailable' },
   458→   458→    smartMoney: { direction: isFr ? 'NEUTRE' : 'NEUTRAL', confidence: isFr ? 'faible' : 'low' },
   459→   459→    retail: { direction: isFr ? 'NEUTRE' : 'NEUTRAL', confidence: isFr ? 'faible' : 'low' },
   460→   460→    contrarianSignal: isFr ? 'Données insuffisantes pour un signal contrarien' : 'Insufficient data for contrarian signal',
   461→   461→    overallSentiment: 'NEUTRAL',
   462→   462→    interpretation: isFr ? 'Données de sentiment temporairement indisponibles. Veuillez réessayer dans quelques minutes.' : 'Sentiment data temporarily unavailable. Please try again in a few minutes.',
   463→   463→    updatedAt: new Date().toISOString(),
   464→   464→  };
   465→   465→
   466→   466→  // 1. Try ZAI SDK first
   467→   467→  const zaiResult = await fetchSentimentWithZAI(lang, period);
   468→   468→  if (zaiResult) {
   469→   469→    const result: any = { ...zaiResult };
   470→   470→    // Compute asset-specific sentiment if asset is provided
   471→   471→    if (asset) {
   472→   472→      // Fetch RSS titles for asset-specific analysis
   473→   473→      try {
   474→   474→        const titles = await fetchSentimentRSS();
   475→   475→        if (titles.length > 0) {
   476→   476→          result.assetSentiment = computeAssetSentiment(titles.join(' '), asset, lang);
   477→   477→        }
   478→   478→      } catch {}
   479→   479→    }
   480→   480→    return result;
   481→   481→  }
   482→   482→

   485→   485→  try {
   486→   486→    const titles = await fetchSentimentRSS();
   487→   487→    if (titles.length > 0) {
   488→   488→      const analysis = analyzeSentimentFromNews(titles, lang);
   489→   489→      const result: any = {
   490→   490→        ...analysis,
   491→   491→        updatedAt: new Date().toISOString(),
   492→   492→      };
   493→   493→      // Add asset-specific sentiment if requested
   494→   494→      if (asset) {
   495→   495→        result.assetSentiment = computeAssetSentiment(titles.join(' '), asset, lang);
   496→   496→      }
   497→   497→      return result;
   498→   498→    }
   499→   499→  } catch (error) {
   500→   500→    console.error('Sentiment RSS fallback error:', error instanceof Error ? error.message : 'Unknown error');
   501→   501→  }
   502→   502→
   503→   503→  // 3. Both failed
   504→   504→  const failResult: any = { ...fallbackResult, error: isFr ? 'Aucune donnée de sentiment disponible' : 'No sentiment data available' };
   505→   505→  if (asset) {
   506→   506→    failResult.assetSentiment = {
   507→   507→      direction: isFr ? 'NEUTRE' : 'NEUTRAL',
   508→   508→      confidence: isFr ? 'faible' : 'low',
   509→   509→      interpretation: isFr ? `Données insuffisantes pour ${asset}.` : `Insufficient data for ${asset}.`,
   510→   510→    };
   511→   511→  }
   512→   512→  return failResult;
   513→   513→}
