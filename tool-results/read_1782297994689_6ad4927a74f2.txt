    43→    43→interface TradeDetail {

    46→    46→  date: string;
    47→    47→  pair: string;
    48→    48→  direction: string;
    49→    49→  session: string;
    50→    50→  marketCondition: string;
    51→    51→  timeframe: string;
    52→    52→  setup: string | null;
    53→    53→  structure: string | null;
    54→    54→  entryModel: string | null;
    55→    55→  amountToWin: number | null;
    56→    56→  entryPrice: number;
    57→    57→  stopLoss: number;
    58→    58→  takeProfit: number;
    59→    59→  exitPrice: number | null;
    60→    60→  entryTime: string | null;
    61→    61→  exitTime: string | null;
    62→    62→  duration: string | null;

    76→    76→
    77→    77→export function TradeDetailDialog() {
    78→    78→  const { selectedTradeId, showTradeDetail, setShowTradeDetail, language, setScreenshotViewerUrl, screenshotViewerUrl } = useAppStore();
    79→    79→  const { trades: allTrades, refetch: refetchTrades } = useTrades();
    80→    80→  const [trade, setTrade] = useState<TradeDetail | null>(null);
    81→    81→  const [loading, setLoading] = useState(false);
    82→    82→  const [error, setError] = useState<string | null>(null);
    83→    83→  const prevTradeIdRef = useRef<string | null>(null);
    84→    84→
    85→    85→  const fetchTrade = useCallback(async (id: string) => {
    86→    86→    setLoading(true);
    87→    87→    setError(null);
    88→    88→    try {
    89→    89→      const res = await fetch(`/api/trades/${id}?_t=${Date.now()}`);
    90→    90→      if (!res.ok) {
    91→    91→        const data = await res.json();
    92→    92→        setError(data.error || (language === "fr" ? "Erreur lors du chargement" : "Error loading trade"));
    93→    93→        setTrade(null);
    94→    94→        return;
    95→    95→      }
    96→    96→      const data = await res.json();
    97→    97→      if (data.trade && data.trade.entryPrice !== undefined) {
    98→    98→        setTrade(data.trade);
    99→    99→      } else {
   100→   100→        setError(language === "fr" ? "Données du trade invalides" : "Invalid trade data");
   101→   101→        setTrade(null);
   102→   102→      }
   103→   103→    } catch (err) {
   104→   104→      console.error("Error fetching trade:", err);
   105→   105→      setError(language === "fr" ? "Erreur de connexion" : "Connection error");
   106→   106→      setTrade(null);
   107→   107→    } finally {
   108→   108→      setLoading(false);
   109→   109→    }
   110→   110→  }, [language]);
   111→   111→
   112→   112→  const handleOpenChange = useCallback((open: boolean) => {
   113→   113→    if (!open) {
   114→   114→      if (screenshotViewerUrl) {
   115→   115→        // Close screenshot viewer instead of the dialog
   116→   116→        setScreenshotViewerUrl(null);
   117→   117→        return;
   118→   118→      }
   119→   119→      setShowTradeDetail(false);
   120→   120→      setTrade(null);
   121→   121→      setError(null);
   122→   122→      prevTradeIdRef.current = null;
   123→   123→    }
   124→   124→  }, [setShowTradeDetail, screenshotViewerUrl, setScreenshotViewerUrl]);
   125→   125→
   126→   126→  useEffect(() => {
   127→   127→    if (showTradeDetail && selectedTradeId && selectedTradeId !== prevTradeIdRef.current) {
   128→   128→      prevTradeIdRef.current = selectedTradeId;
   129→   129→      setTrade(null);
   130→   130→      fetchTrade(selectedTradeId);
   131→   131→      // Fetch all trades for chart data and screenshot fallback
   132→   132→      refetchTrades();
   133→   133→    }
   134→   134→    if (!showTradeDetail && prevTradeIdRef.current) {
   135→   135→      prevTradeIdRef.current = null;
   136→   136→      setTrade(null);
   137→   137→      setError(null);
   138→   138→    }
   139→   139→  }, [showTradeDetail, selectedTradeId, fetchTrade, refetchTrades]);
   140→   140→
   141→   141→  const isLong = trade?.direction === "LONG";
   142→   142→  const isWin = trade?.result === "WIN";
   143→   143→  const isLoss = trade?.result === "LOSS";
   144→   144→
   145→   145→  // Compute trade statistics
   146→   146→  const tradeStats = useMemo(() => {
   147→   147→    if (!trade) return null;
   148→   148→    const risk = Math.abs(trade.entryPrice - trade.stopLoss);
   149→   149→    const reward = Math.abs(trade.takeProfit - trade.entryPrice);
   150→   150→    const rr = risk > 0 ? reward / risk : 0;
   151→   151→    const riskRewardRatio = risk > 0 ? rr.toFixed(2) : "—";
   152→   152→
   153→   153→    // Recalculate P&L FIRST if it's 0/null but we have the data to compute it
   154→   154→    let pnl = trade.pnl;
   155→   155→    if ((pnl === null || pnl === 0) && trade.exitPrice != null && trade.lotSize && trade.lotSize > 0) {
   156→   156→      const priceDiff = trade.direction === 'LONG'
   157→   157→        ? trade.exitPrice - trade.entryPrice
   158→   158→        : trade.entryPrice - trade.exitPrice;
   159→   159→      pnl = calculateDollarAmount(priceDiff, trade.lotSize, trade.pair);
   160→   160→    }
   161→   161→
   162→   162→    // Calculate Risk ($) and Reward ($) using contract size
   163→   163→    let riskAmount = "—";
   164→   164→    let rewardAmount = "—";
   165→   165→    let maxRewardDollar = 0;
   166→   166→
   167→   167→    if (trade.amountToWin && trade.amountToWin > 0) {
   168→   168→      // Use user-entered amountToWin for reward, derive risk from RR
   169→   169→      rewardAmount = trade.amountToWin.toFixed(2);
   170→   170→      riskAmount = rr > 0 ? (trade.amountToWin / rr).toFixed(2) : trade.amountToWin.toFixed(2);
   171→   171→      maxRewardDollar = trade.amountToWin;
   172→   172→    } else if (trade.lotSize && trade.lotSize > 0) {
   173→   173→      // Calculate using contract size based on pair
   174→   174→      const contractSize = getContractSize(trade.pair);
   175→   175→      const riskDollar = risk * trade.lotSize * contractSize;
   176→   176→      const rewardDollar = reward * trade.lotSize * contractSize;
   177→   177→      riskAmount = riskDollar.toFixed(2);
   178→   178→      rewardAmount = rewardDollar.toFixed(2);
   179→   179→      maxRewardDollar = rewardDollar;
   180→   180→    }
   181→   181→
   182→   182→    // Efficiency: actual P&L as percentage of max possible reward
   183→   183→    let efficiency = "—";
   184→   184→    if (pnl != null && maxRewardDollar > 0) {
   185→   185→      const eff = (pnl / maxRewardDollar) * 100;
   186→   186→      efficiency = eff.toFixed(0) + "%";
   187→   187→    }
   188→   188→
   189→   189→    return { riskRewardRatio, riskAmount, rewardAmount, efficiency, pnl };
   190→   190→  }, [trade]);
   191→   191→
   192→   192→  // Get chart data for this trade
   193→   193→  const chartData = useMemo(() => {
   194→   194→    if (!trade || !allTrades.length) return { last10RR: [], cumulativeRRLast10: [], totalRRLast10: 0, cumulativeRRUpToTrade: [], totalCumulativeRR: 0 };
   195→   195→
   196→   196→    // Sort trades by date then createdAt
   197→   197→    const sorted = [...allTrades].sort((a, b) => {
   198→   198→      const dateA = new Date(a.date).getTime();
   199→   199→      const dateB = new Date(b.date).getTime();
   200→   200→      if (dateA !== dateB) return dateA - dateB;
   201→   201→      return new Date(a.createdAt).getTime() - new Date(b.createdAt).getTime();
   202→   202→    });
   203→   203→
   204→   204→    // Find the index of current trade
   205→   205→    const currentIdx = sorted.findIndex(t => t.id === trade.id);
   206→   206→
   207→   207→    // ─── RR PAR TRADE: Last 10 trades ending at this trade ───
   208→   208→    const startIdx = Math.max(0, (currentIdx >= 0 ? currentIdx : sorted.length - 1) - 9);
   209→   209→    const endIdx = currentIdx >= 0 ? currentIdx + 1 : sorted.length;
   210→   210→    const last10 = sorted.slice(startIdx, endIdx);
   211→   211→
   212→   212→    const last10RR = last10.map((t, i) => ({
   213→   213→      index: startIdx + i + 1,
   214→   214→      rr: t.rr ?? 0,
   215→   215→      result: t.result,
   216→   216→    }));
   217→   217→
   218→   218→    // ─── CUMULE RR ISOLÉ: Cumulative RR of last 10 trades ───
   219→   219→    let cumRRLast10 = 0;
   220→   220→    const cumulativeRRLast10 = last10.map((t, i) => {
   221→   221→      cumRRLast10 += (t.rr ?? 0);
   222→   222→      return { index: i + 1, cumulativeRR: cumRRLast10 };
   223→   223→    });
   224→   224→    const totalRRLast10 = cumRRLast10;
   225→   225→
   226→   226→    // ─── CUMULE TOTAL DE RR: Cumulative RR from first trade up to this trade ───
   227→   227→    const upToCurrentIdx = currentIdx >= 0 ? currentIdx + 1 : sorted.length;
   228→   228→    const tradesUpToCurrent = sorted.slice(0, upToCurrentIdx);
   229→   229→
   230→   230→    let cumRR = 0;
   231→   231→    const cumulativeRRUpToTrade = tradesUpToCurrent.map((t, i) => {
   232→   232→      cumRR += (t.rr ?? 0);
   233→   233→      return { index: i + 1, cumulativeRR: cumRR };
   234→   234→    });
   235→   235→
   236→   236→    const totalCumulativeRR = cumRR;
   237→   237→
   238→   238→    return { last10RR, cumulativeRRLast10, totalRRLast10, cumulativeRRUpToTrade, totalCumulativeRR };
   239→   239→  }, [trade, allTrades]);
   240→   240→
   241→   241→  // Chart dimensions
   242→   242→  const CHART_W = 300;
   243→   243→  const CHART_H = 120;
   244→   244→  const CHART_PAD = 20;
   245→   245→
   246→   246→  // Bar chart data for RR PAR TRADE
   247→   247→  const rrBarChart = useMemo(() => {
   248→   248→    const data = chartData.last10RR;
   249→   249→    if (!data.length) return null;
   250→   250→    const maxRR = Math.max(...data.map(d => Math.abs(d.rr)), 1);
   251→   251→    const barW = Math.max(4, (CHART_W - CHART_PAD * 2) / data.length - 4);
   252→   252→    const zeroY = CHART_H / 2;
   253→   253→
   254→   254→    return (
   255→   255→      <svg width="100%" height={CHART_H} viewBox={`0 0 ${CHART_W} ${CHART_H}`} className="overflow-visible">
   256→   256→        {/* Zero line */}
   257→   257→        <line x1={CHART_PAD} y1={zeroY} x2={CHART_W - CHART_PAD} y2={zeroY} stroke="currentColor" strokeOpacity={0.15} strokeDasharray="3,3" />
   258→   258→        {data.map((d, i) => {
   259→   259→          const x = CHART_PAD + i * ((CHART_W - CHART_PAD * 2) / data.length) + 2;
   260→   260→          const barHeight = (Math.abs(d.rr) / maxRR) * (zeroY - CHART_PAD);
   261→   261→          const y = d.rr >= 0 ? zeroY - barHeight : zeroY;
   262→   262→          const color = d.result === "WIN" ? "#22c55e" : d.result === "LOSS" ? "#ef4444" : "#eab308";
   263→   263→          return (
   264→   264→            <g key={i}>
   265→   265→              <rect x={x} y={y} width={barW} height={barHeight} fill={color} rx={2} opacity={0.8} />
   266→   266→              <text x={x + barW / 2} y={CHART_H - 2} textAnchor="middle" fontSize={8} fill="currentColor" opacity={0.4}>{d.index}</text>
   267→   267→            </g>
   268→   268→          );
   269→   269→        })}
   270→   270→      </svg>
   271→   271→    );
   272→   272→  }, [chartData.last10RR]);
   273→   273→
   274→   274→  // Line chart for CUMULE RR ISOLÉ (last 10 trades)
   275→   275→  const cumulativeLineChart = useMemo(() => {
   276→   276→    const data = chartData.cumulativeRRLast10;
   277→   277→    if (!data.length) return null;
   278→   278→    const values = data.map(d => d.cumulativeRR);
   279→   279→    const minVal = Math.min(...values, 0);
   280→   280→    const maxVal = Math.max(...values, 0);
   281→   281→    const range = maxVal - minVal || 1;
   282→   282→
   283→   283→    const chartWidth = CHART_W - CHART_PAD * 2;
   284→   284→    const chartHeight = CHART_H - CHART_PAD * 2;
   285→   285→
   286→   286→    const points = data.map((d, i) => ({
   287→   287→      x: CHART_PAD + (data.length === 1 ? chartWidth / 2 : i * (chartWidth / (data.length - 1))),
   288→   288→      y: CHART_H - CHART_PAD - ((d.cumulativeRR - minVal) / range) * chartHeight,
   289→   289→    }));
   290→   290→
   291→   291→    const pathD = points.map((p, i) => `${i === 0 ? "M" : "L"} ${p.x} ${p.y}`).join(" ");
   292→   292→    const areaD = data.length > 1
   293→   293→      ? `${pathD} L ${points[points.length - 1].x} ${CHART_H - CHART_PAD} L ${points[0].x} ${CHART_H - CHART_PAD} Z`
   294→   294→      : `M ${points[0].x} ${CHART_H - CHART_PAD} L ${points[0].x} ${points[0].y} Z`;
   295→   295→
   296→   296→    // Zero line Y
   297→   297→    const zeroY = CHART_H - CHART_PAD - ((0 - minVal) / range) * chartHeight;
   298→   298→
   299→   299→    // Determine color based on final cumulative RR
   300→   300→    const finalRR = values[values.length - 1];
   301→   301→    const lineColor = finalRR >= 0 ? "#22c55e" : "#ef4444";
   302→   302→
   303→   303→    // Total RR for display
   304→   304→    const totalRR = chartData.totalRRLast10;
   305→   305→    const totalRRColor = totalRR >= 0 ? "#22c55e" : "#ef4444";
   306→   306→
   307→   307→    return (
   308→   308→      <svg width="100%" height={CHART_H} viewBox={`0 0 ${CHART_W} ${CHART_H}`} className="overflow-visible">
   309→   309→        {zeroY > CHART_PAD && zeroY < CHART_H - CHART_PAD && (
   310→   310→          <line x1={CHART_PAD} y1={zeroY} x2={CHART_W - CHART_PAD} y2={zeroY} stroke="currentColor" strokeOpacity={0.15} strokeDasharray="3,3" />
   311→   311→        )}
   312→   312→        <path d={areaD} fill={lineColor} opacity={0.1} />
   313→   313→        {data.length > 1 && (
   314→   314→          <path d={pathD} fill="none" stroke={lineColor} strokeWidth={2} />
   315→   315→        )}
   316→   316→        {points.map((p, i) => (
   317→   317→          <circle key={i} cx={p.x} cy={p.y} r={3} fill={lineColor} />
   318→   318→        ))}
   319→   319→        {/* Total RR display at top-right corner */}
   320→   320→        <text
   321→   321→          x={CHART_W - CHART_PAD}
   322→   322→          y={CHART_PAD - 4}
   323→   323→          textAnchor="end"
   324→   324→          fontSize={11}
   325→   325→          fontWeight="bold"
   326→   326→          fill={totalRRColor}
   327→   327→        >
   328→   328→          {totalRR >= 0 ? "+" : ""}{totalRR.toFixed(2)} RR
   329→   329→        </text>
   330→   330→      </svg>
   331→   331→    );
   332→   332→  }, [chartData.cumulativeRRLast10, chartData.totalRRLast10]);
   333→   333→
   334→   334→  // Resolve screenshot URL - handles all possible URL formats
   335→   335→  const resolveScreenshotUrl = useCallback((url: string): string => {
   336→   336→    if (!url) return '';
   337→   337→    // Full URLs (Cloudinary, etc.) used directly
   338→   338→    if (url.startsWith('http://') || url.startsWith('https://')) return url;
   339→   339→    // Local paths get converted to API route
   340→   340→    if (url.startsWith('upload/screenshots/')) {
   341→   341→      return `/api/screenshots/${url.replace('upload/screenshots/', '')}`;
   342→   342→    }
   343→   343→    // Any other path - try as API route
   344→   344→    return `/api/screenshots/${url.split('/').pop()}`;
   345→   345→  }, []);
   346→   346→
   347→   347→  // Detect if a URL is a video (Cloudinary video URL or video file extension)
   348→   348→  const isVideoUrl = useCallback((url: string): boolean => {
   349→   349→    if (!url) return false;
   350→   350→    // Cloudinary video URLs contain /video/upload/
   351→   351→    if (url.includes('/video/upload/')) return true;
   352→   352→    // Check video file extensions
   353→   353→    const videoExts = ['.mp4', '.webm', '.mov', '.avi', '.mkv', '.m4v'];
   354→   354→    const lowerUrl = url.toLowerCase();
   355→   355→    return videoExts.some(ext => lowerUrl.endsWith(ext));
   356→   356→  }, []);
   357→   357→
   358→   358→  // Get screenshots - check both trade API and allTrades for maximum reliability
   359→   359→  const tradeScreenshots = useMemo(() => {
   360→   360→    if (!trade) return [];
   361→   361→    // Try allTrades first (freshly refetched when dialog opens, includes screenshots)
   362→   362→    const fromAllTrades = allTrades.find(t => t.id === trade.id)?.screenshots;
   363→   363→    if (fromAllTrades && fromAllTrades.length > 0) return fromAllTrades;
   364→   364→    // Fallback: trade API data
   365→   365→    if (trade.screenshots && trade.screenshots.length > 0) return trade.screenshots;
   366→   366→    return [];
   367→   367→  }, [trade, allTrades]);
   368→   368→
   369→   369→  return (
   370→   370→    <Dialog open={showTradeDetail} onOpenChange={handleOpenChange}>
   371→   371→      <DialogContent className="sm:max-w-5xl max-h-[90vh] p-0 gap-0 overflow-hidden">
   372→   372→        {/* Header */}
   373→   373→        {loading ? (
   374→   374→          <div className="p-6 space-y-4">
   375→   375→            <DialogHeader>
   376→   376→              <DialogTitle className="sr-only">{language === "fr" ? "Chargement du trade..." : "Loading trade..."}</DialogTitle>
   377→   377→              <DialogDescription className="sr-only">{language === "fr" ? "Chargement en cours" : "Loading"}</DialogDescription>
   378→   378→            </DialogHeader>
   379→   379→            <Skeleton className="h-8 w-48" />
   380→   380→            <Skeleton className="h-4 w-64" />
   381→   381→          </div>
   382→   382→        ) : error ? (
   383→   383→          <div className="p-6">
   384→   384→            <DialogHeader>
   385→   385→              <DialogTitle className="sr-only">{language === "fr" ? "Erreur" : "Error"}</DialogTitle>
   386→   386→              <DialogDescription className="sr-only">{error}</DialogDescription>
   387→   387→            </DialogHeader>
   388→   388→            <div className="flex flex-col items-center justify-center py-12 text-center">
   389→   389→              <AlertTriangle className="w-12 h-12 text-loss/50 mb-4" />
   390→   390→              <p className="text-muted-foreground text-sm">{error}</p>
   391→   391→              <Button variant="outline" size="sm" className="mt-4" onClick={() => handleOpenChange(false)}>
   392→   392→                {language === "fr" ? "Fermer" : "Close"}
   393→   393→              </Button>
   394→   394→            </div>
   395→   395→          </div>
   396→   396→        ) : trade ? (
   397→   397→          <>
   398→   398→            {/* Trade Header */}
   399→   399→            <div className="p-6 border-b border-border bg-muted/30">
   400→   400→              <DialogHeader>
   401→   401→                <DialogTitle className="flex items-center gap-3 text-xl">
   402→   402→                  <div className={cn(
   403→   403→                    "w-10 h-10 rounded-xl flex items-center justify-center",
   404→   404→                    isLong ? "bg-profit/10 text-profit" : "bg-loss/10 text-loss"
   405→   405→                  )}>
   406→   406→                    {isLong ? <ArrowUpRight className="w-5 h-5" /> : <ArrowDownRight className="w-5 h-5" />}
   407→   407→                  </div>
   408→   408→                  <div className="flex items-center gap-2 flex-wrap">
   409→   409→                    <span className="font-mono font-bold text-lg">{trade.pair}</span>
   410→   410→                    <DialogDescription className="sr-only">{trade.pair} {trade.direction} - {trade.date}</DialogDescription>
   411→   411→                    <Badge className={cn(
   412→   412→                      "text-xs font-semibold",
   413→   413→                      isLong ? "bg-profit/15 text-profit border-profit/20" : "bg-loss/15 text-loss border-loss/20"
   414→   414→                    )}>
