    94→    94→interface DayTradeData {
    95→    95→  date: Date;
    96→    96→  trades: TradeData[];
    97→    97→  totalRR: number;
    98→    98→  totalPnL: number;
    99→    99→  isProfit: boolean | null; // null = no trades
   100→   100→}
   101→   101→
   102→   102→interface TradeData {
   103→   103→  id: string;
   104→   104→  userId: string;
   105→   105→  date: string;
   106→   106→  pair: string;
   107→   107→  direction: string;
   108→   108→  session: string;
   109→   109→  marketCondition: string;
   110→   110→  timeframe: string;
   111→   111→  setup: string | null;
   112→   112→  entryPrice: number;
   113→   113→  stopLoss: number;
   114→   114→  takeProfit: number;
   115→   115→  exitPrice: number | null;
   116→   116→  entryTime: string | null;
   117→   117→  exitTime: string | null;
   118→   118→  duration: string | null;
   119→   119→  lotSize: number | null;
   120→   120→  rr: number | null;
   121→   121→  pnl: number | null;
   122→   122→  result: string | null;
   123→   123→  newsEnabled: boolean;
   124→   124→  emotions: string | null;
   125→   125→  confluence: string | null;
   126→   126→  mistakes: string | null;
   127→   127→  lessons: string | null;
   128→   128→  notes: string | null;
   129→   129→  createdAt: string;
   130→   130→  screenshots: { id: string; type: string; url: string }[];
   131→   131→}

   145→   145→// ─── Main Component ──────────────────────────────────────────
   146→   146→export function JournalTab() {
   147→   147→  const { user, language, setScreenshotViewerUrl, setSelectedTradeId, setShowTradeDetail } = useAppStore();
   148→   148→  const { period, setPeriod, filters } = useTimeFilter();
   149→   149→  const { trades, loading, refetch } = useTrades(filters);
   150→   150→
   151→   151→  // Fetch on mount
   152→   152→  React.useEffect(() => { refetch(); }, [refetch]);
   153→   153→  const isMobile = useIsMobile();
   154→   154→  const isAdminUser = user?.role === "admin";
   155→   155→
   156→   156→  const [currentMonth, setCurrentMonth] = useState(new Date());
   157→   157→  const [selectedDate, setSelectedDate] = useState<Date | null>(null);
   158→   158→  const [expandedTradeId, setExpandedTradeId] = useState<string | null>(null);
   159→   159→  const [mobileDetailOpen, setMobileDetailOpen] = useState(false);
   160→   160→
   161→   161→  // Delete confirmation state
   162→   162→  const [deleteTarget, setDeleteTarget] = useState<{ id: string; pair: string; direction: string } | null>(null);
   163→   163→  const [isDeleting, setIsDeleting] = useState(false);
   164→   164→
   165→   165→  const handleDeleteTrade = useCallback(async () => {
   166→   166→    if (!deleteTarget || isDeleting) return;
   167→   167→    setIsDeleting(true);
   168→   168→    try {
   169→   169→      const res = await fetch(`/api/trades/${deleteTarget.id}`, { method: "DELETE" });
   170→   170→      if (!res.ok) {
   171→   171→        const data = await res.json();
   172→   172→        throw new Error(data.error || "Delete failed");
   173→   173→      }
   174→   174→      toast.success(language === "fr" ? "Trade supprimé !" : "Trade deleted!");
   175→   175→      setDeleteTarget(null);
   176→   176→      refetch();
   177→   177→    } catch (error: any) {
   178→   178→      toast.error(error.message || (language === "fr" ? "Erreur lors de la suppression" : "Delete failed"));
   179→   179→    } finally {
   180→   180→      setIsDeleting(false);
   181→   181→    }
   182→   182→  }, [deleteTarget, isDeleting, language, refetch]);
   183→   183→
   184→   184→  // Group trades by date
   185→   185→  const tradesByDate = useMemo(() => {
   186→   186→    const map = new Map<string, DayTradeData>();
   187→   187→
   188→   188→    for (const trade of trades) {
   189→   189→      const tradeDate = parseISO(trade.date);
   190→   190→      if (!isValid(tradeDate)) continue;
   191→   191→
   192→   192→      const key = format(tradeDate, "yyyy-MM-dd");
   193→   193→
   194→   194→      if (!map.has(key)) {
   195→   195→        map.set(key, {
   196→   196→          date: tradeDate,
   197→   197→          trades: [],
   198→   198→          totalRR: 0,
   199→   199→          totalPnL: 0,
   200→   200→          isProfit: null,
   201→   201→        });
   202→   202→      }
   203→   203→
   204→   204→      const dayData = map.get(key)!;
   205→   205→      dayData.trades.push(trade as TradeData);
   206→   206→      if (trade.rr !== null) dayData.totalRR += trade.rr;
   207→   207→      if (trade.pnl !== null) dayData.totalPnL += trade.pnl;
   208→   208→    }
   209→   209→
   210→   210→    // Determine profit/loss for each day
   211→   211→    for (const [, dayData] of map) {
   212→   212→      if (dayData.trades.length > 0) {
   213→   213→        dayData.isProfit = dayData.totalPnL >= 0;
   214→   214→      }
   215→   215→    }
   216→   216→
   217→   217→    return map;
   218→   218→  }, [trades]);
   219→   219→
   220→   220→  // Calendar grid data
   221→   221→  const calendarDays = useMemo(() => {
   222→   222→    const monthStart = startOfMonth(currentMonth);
   223→   223→    const monthEnd = endOfMonth(currentMonth);
   224→   224→    const calendarStart = startOfWeek(monthStart, { weekStartsOn: 0 }); // Sunday
   225→   225→    const calendarEnd = endOfWeek(monthEnd, { weekStartsOn: 0 });
   226→   226→
   227→   227→    return eachDayOfInterval({ start: calendarStart, end: calendarEnd });
   228→   228→  }, [currentMonth]);
   229→   229→
   230→   230→  // Selected date trades
   231→   231→  const selectedDayTrades = useMemo(() => {
   232→   232→    if (!selectedDate) return null;
   233→   233→    const key = format(selectedDate, "yyyy-MM-dd");
   234→   234→    return tradesByDate.get(key) || null;
   235→   235→  }, [selectedDate, tradesByDate]);
   236→   236→
   237→   237→  // Navigate months
   238→   238→  const goToPreviousMonth = useCallback(() => {
   239→   239→    setCurrentMonth((prev) => subMonths(prev, 1));
   240→   240→  }, []);
   241→   241→
   242→   242→  const goToNextMonth = useCallback(() => {
   243→   243→    setCurrentMonth((prev) => addMonths(prev, 1));
   244→   244→  }, []);
   245→   245→
   246→   246→  // Handle date click
   247→   247→  const handleDateClick = useCallback(
   248→   248→    (date: Date) => {
   249→   249→      const key = format(date, "yyyy-MM-dd");
   250→   250→      const dayData = tradesByDate.get(key);
   251→   251→      if (dayData && dayData.trades.length > 0) {
   252→   252→        setSelectedDate(date);
   253→   253→        setExpandedTradeId(null);
   254→   254→        if (isMobile) {
   255→   255→          setMobileDetailOpen(true);
   256→   256→        }
   257→   257→      }
   258→   258→    },
   259→   259→    [tradesByDate, isMobile]
   260→   260→  );
   261→   261→
   262→   262→  // Format date header in French
   263→   263→  const formatDateHeader = useCallback(
   264→   264→    (date: Date) => {
   265→   265→      const dayName = FRENCH_DAY_NAMES[getDay(date)];
   266→   266→      const day = format(date, "d");
   267→   267→      const month = FRENCH_MONTHS[date.getMonth()];
   268→   268→      const year = date.getFullYear();
   269→   269→      return `${dayName} ${day} ${month} ${year}`.toUpperCase();
   270→   270→    },
   271→   271→    []
   272→   272→  );
   273→   273→
   274→   274→  // Handle trade click for detail view
   275→   275→  const handleTradeDetailClick = useCallback(
   276→   276→    (tradeId: string) => {
   277→   277→      setSelectedTradeId(tradeId);
   278→   278→      setShowTradeDetail(true);
   279→   279→    },
   280→   280→    [setSelectedTradeId, setShowTradeDetail]
   281→   281→  );
   282→   282→
   283→   283→  // Trade detail panel content
   284→   284→  const tradeDetailContent = useMemo(() => {
   285→   285→    if (!selectedDate || !selectedDayTrades) {
   286→   286→      return (
   287→   287→        <div className="flex flex-col items-center justify-center h-full min-h-[300px] text-center p-6">
   288→   288→          <CalendarIcon className="w-16 h-16 text-muted-foreground/20 mb-4" />
   289→   289→          <p className="text-muted-foreground text-sm">
   290→   290→            {t(language, "clickDate")}
   291→   291→          </p>
   292→   292→        </div>
   293→   293→      );
   294→   294→    }
   295→   295→
   296→   296→    const dayTotalRR = selectedDayTrades.totalRR;
   297→   297→    const dayTotalPnL = selectedDayTrades.totalPnL;
   298→   298→
   299→   299→    return (
   300→   300→      <div className="space-y-4">
   301→   301→        {/* Date Header */}
   302→   302→        <div className="space-y-2">
   303→   303→          <h3 className="text-sm font-bold tracking-wider text-foreground/80">
   304→   304→            {formatDateHeader(selectedDate)}
   305→   305→          </h3>
   306→   306→          <div className="flex items-center gap-3">
   307→   307→            <Badge
   308→   308→              variant="outline"
   309→   309→              className={cn(
   310→   310→                "text-xs font-mono gap-1",
   311→   311→                dayTotalPnL >= 0
   312→   312→                  ? "text-profit border-profit/30 bg-profit/5"
   313→   313→                  : "text-loss border-loss/30 bg-loss/5"
   314→   314→              )}
   315→   315→            >
   316→   316→              {dayTotalPnL >= 0 ? (
   317→   317→                <TrendingUp className="w-3 h-3" />
   318→   318→              ) : (
   319→   319→                <TrendingDown className="w-3 h-3" />
   320→   320→              )}
   321→   321→              P&L: {dayTotalPnL >= 0 ? "+" : ""}
   322→   322→              {dayTotalPnL.toFixed(2)}
   323→   323→            </Badge>
   324→   324→            <Badge variant="outline" className="text-xs font-mono gap-1">
   325→   325→              <Target className="w-3 h-3" />
   326→   326→              RR: {dayTotalRR.toFixed(2)}
   327→   327→            </Badge>
   328→   328→            <Badge variant="secondary" className="text-xs">
   329→   329→              {selectedDayTrades.trades.length}{" "}
   330→   330→              {selectedDayTrades.trades.length > 1
   331→   331→                ? language === "fr"
   332→   332→                  ? "trades"
   333→   333→                  : "trades"
   334→   334→                : "trade"}
   335→   335→            </Badge>
   336→   336→          </div>
   337→   337→        </div>
   338→   338→
   339→   339→        <Separator />
   340→   340→
   341→   341→        {/* Trade Cards */}
   342→   342→        <div className="space-y-3">
   343→   343→          {selectedDayTrades.trades.map((trade, index) => (
   344→   344→            <TradeCard
   345→   345→              key={trade.id}
   346→   346→              trade={trade}
   347→   347→              index={index}
   348→   348→              language={language}
   349→   349→              isAdmin={isAdminUser}
   350→   350→              isExpanded={expandedTradeId === trade.id}
   351→   351→              onToggleExpand={() =>
   352→   352→                setExpandedTradeId((prev) =>
   353→   353→                  prev === trade.id ? null : trade.id
   354→   354→                )
   355→   355→              }
   356→   356→              onScreenshotClick={setScreenshotViewerUrl}
   357→   357→              onTradeDetailClick={handleTradeDetailClick}
   358→   358→              onDeleteClick={(id) => setDeleteTarget({ id, pair: trade.pair, direction: trade.direction })}
   359→   359→            />
   360→   360→          ))}
   361→   361→        </div>
   362→   362→
   363→   363→        {/* Mini Charts */}
   364→   364→        {selectedDayTrades.trades.length > 1 && (
   365→   365→          <>
   366→   366→            <Separator />
   367→   367→            <MiniCharts
   368→   368→              trades={selectedDayTrades.trades}
   369→   369→              language={language}
   370→   370→            />
   371→   371→          </>
   372→   372→        )}
   373→   373→      </div>
   374→   374→    );
   375→   375→  }, [
   376→   376→    selectedDate,
   377→   377→    selectedDayTrades,
   378→   378→    language,
   379→   379→    expandedTradeId,
   380→   380→    formatDateHeader,
   381→   381→    setScreenshotViewerUrl,
   382→   382→    handleTradeDetailClick,
   383→   383→    isAdminUser,
   384→   384→    deleteTarget,
   385→   385→    isDeleting,
   386→   386→    handleDeleteTrade,
   387→   387→  ]);
   388→   388→
   389→   389→  if (loading) {
   390→   390→    return (
   391→   391→      <div className="p-4 md:p-6 max-w-[1600px] mx-auto overflow-x-hidden">
   392→   392→        <div className="flex flex-col lg:flex-row gap-6">
   393→   393→          <div className="flex-1 lg:flex-[3]">
   394→   394→            <Card className="p-6">
   395→   395→              <Skeleton className="h-8 w-48 mb-6" />
   396→   396→              <div className="grid grid-cols-7 gap-2">
   397→   397→                {Array.from({ length: 35 }).map((_, i) => (
   398→   398→                  <Skeleton key={i} className="h-16 w-full rounded-lg" />
   399→   399→                ))}
   400→   400→              </div>
   401→   401→            </Card>
   402→   402→          </div>
   403→   403→          <div className="flex-1 lg:flex-[2]">
   404→   404→            <Card className="p-6 h-80">
   405→   405→              <Skeleton className="h-full w-full" />
   406→   406→            </Card>
   407→   407→          </div>
   408→   408→        </div>
   409→   409→      </div>
   410→   410→    );
   411→   411→  }
   412→   412→
   413→   413→  return (
   414→   414→    <div className="p-4 md:p-6 max-w-[1600px] mx-auto overflow-x-hidden">
   415→   415→      <div className="flex flex-col lg:flex-row gap-4 md:gap-6">
   416→   416→        {/* Calendar Section (Left - 60%) */}
   417→   417→        <div className="flex-1 lg:flex-[3]">
   418→   418→          <Card className="overflow-hidden">
   419→   419→            {/* Calendar Header */}
   420→   420→            <div className="p-4 md:p-6 border-b border-border space-y-3">
   421→   421→              <div className="flex items-center justify-between">
   422→   422→                <h2 className="text-lg font-bold tracking-tight text-foreground">
   423→   423→                  {FRENCH_MONTHS[currentMonth.getMonth()]}{" "}
   424→   424→                  {currentMonth.getFullYear()}
   425→   425→                </h2>
   426→   426→                <div className="flex items-center gap-1">
   427→   427→                  <Button
   428→   428→                    variant="ghost"
   429→   429→                    size="icon"
   430→   430→                    onClick={goToPreviousMonth}
   431→   431→                    className="h-8 w-8"
   432→   432→                  >
   433→   433→                    <ChevronLeft className="w-4 h-4" />
   434→   434→                  </Button>
   435→   435→                  <Button
   436→   436→                    variant="ghost"
   437→   437→                    size="icon"
   438→   438→                    onClick={goToNextMonth}
   439→   439→                    className="h-8 w-8"
   440→   440→                  >
   441→   441→                    <ChevronRight className="w-4 h-4" />
   442→   442→                  </Button>
   443→   443→                </div>
   444→   444→              </div>
   445→   445→              {/* Time Filter */}
   446→   446→              <TimeFilterBar language={language} period={period} onPeriodChange={setPeriod} />
   447→   447→            </div>
   448→   448→
   449→   449→            {/* Calendar Grid */}
   450→   450→            <div className="p-3 md:p-4">
   451→   451→              {/* Day Headers */}
   452→   452→              <div className="grid grid-cols-7 mb-2">
   453→   453→                {FRENCH_DAYS.map((day) => (
   454→   454→                  <div
   455→   455→                    key={day}
   456→   456→                    className="text-center text-xs font-semibold text-muted-foreground uppercase tracking-wider py-1"
   457→   457→                  >
   458→   458→                    {day}
   459→   459→                  </div>
   460→   460→                ))}
   461→   461→              </div>
   462→   462→
   463→   463→              {/* Day Cells */}
   464→   464→              <div className="grid grid-cols-7 gap-1">
   465→   465→                {calendarDays.map((day, idx) => {
   466→   466→                  const key = format(day, "yyyy-MM-dd");
   467→   467→                  const dayData = tradesByDate.get(key);
   468→   468→                  const hasTrades = dayData && dayData.trades.length > 0;
   469→   469→                  const isCurrentMonth = isSameMonth(day, currentMonth);
   470→   470→                  const isSelected =
   471→   471→                    selectedDate && isSameDay(day, selectedDate);
   472→   472→                  const isTodayDate = isToday(day);
   473→   473→
   474→   474→                  return (
   475→   475→                    <button
   476→   476→                      key={idx}
   477→   477→                      onClick={() => handleDateClick(day)}
   478→   478→                      disabled={!hasTrades || !isCurrentMonth}
   479→   479→                      className={cn(
   480→   480→                        "relative flex flex-col items-center justify-center rounded-lg p-1 min-h-[44px] md:min-h-[56px] transition-all duration-200 text-sm",
   481→   481→                        // Base styles
   482→   482→                        "hover:bg-muted/50",
   483→   483→                        // Current month vs outside
   484→   484→                        isCurrentMonth
   485→   485→                          ? "text-foreground"
   486→   486→                          : "text-muted-foreground/40",
   487→   487→                        // Today
   488→   488→                        isTodayDate &&
   489→   489→                          !isSelected &&
   490→   490→                          "ring-1 ring-primary/30",
   491→   491→                        // Selected
   492→   492→                        isSelected &&
   493→   493→                          "ring-2 ring-primary bg-primary/10",
   494→   494→                        // Has trades - clickable
   495→   495→                        hasTrades && isCurrentMonth
   496→   496→                          ? "cursor-pointer hover:scale-[1.02]"
   497→   497→                          : "cursor-default",
   498→   498→                        // No trades - dim
