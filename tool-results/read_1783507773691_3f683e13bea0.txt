   117→// ─── Custom Setup Hook ───────────────────────────────────────
   118→function loadCustomSetupsFromStorage(): CustomSetup[] {
   119→  if (typeof window === "undefined") return DEFAULT_CUSTOM_SETUPS;
   120→  try {
   121→    const stored = localStorage.getItem(CUSTOM_SETUPS_STORAGE_KEY);
   122→    if (stored) {
   123→      const parsed = JSON.parse(stored);
   124→      if (Array.isArray(parsed) && parsed.length > 0) {
   125→        return parsed;
   126→      }
   127→    }
   128→  } catch {
   129→    // ignore parse errors
   130→  }
   131→  return DEFAULT_CUSTOM_SETUPS;
   132→}
   133→
   134→function useCustomSetups() {
   135→  const [customSetups, setCustomSetups] = useState<CustomSetup[]>(loadCustomSetupsFromStorage);
   136→
   137→  const saveSetups = useCallback((setups: CustomSetup[]) => {
   138→    setCustomSetups(setups);
   139→    try {
   140→      localStorage.setItem(CUSTOM_SETUPS_STORAGE_KEY, JSON.stringify(setups));
   141→    } catch {
   142→      // ignore storage errors
   143→    }
   144→  }, []);
   145→
   146→  const addSetup = useCallback((setup: Omit<CustomSetup, "id">) => {
   147→    setCustomSetups(prev => {
   148→      const maxNum = prev.reduce((max, s) => {
   149→        const match = s.id.match(/^custom-(\d+)$/);
   150→        return match ? Math.max(max, parseInt(match[1])) : max;
   151→      }, 0);
   152→      const newSetup: CustomSetup = { ...setup, id: `custom-${maxNum + 1}` };
   153→      const updated = [...prev, newSetup];
   154→      try {
   155→        localStorage.setItem(CUSTOM_SETUPS_STORAGE_KEY, JSON.stringify(updated));
   156→      } catch { /* ignore */ }
   157→      return updated;
   158→    });
   159→  }, []);
   160→
   161→  const updateSetup = useCallback((id: string, updates: Partial<Omit<CustomSetup, "id">>) => {
   162→    setCustomSetups(prev => {
   163→      const updated = prev.map(s => s.id === id ? { ...s, ...updates } : s);
   164→      try {
   165→        localStorage.setItem(CUSTOM_SETUPS_STORAGE_KEY, JSON.stringify(updated));
   166→      } catch { /* ignore */ }
   167→      return updated;
   168→    });
   169→  }, []);
   170→
   171→  const deleteSetup = useCallback((id: string) => {
   172→    setCustomSetups(prev => {
   173→      const updated = prev.filter(s => s.id !== id);
   174→      try {
   175→        localStorage.setItem(CUSTOM_SETUPS_STORAGE_KEY, JSON.stringify(updated));
   176→      } catch { /* ignore */ }
   177→      return updated;
   178→    });
   179→  }, []);
   180→
   181→  return { customSetups, addSetup, updateSetup, deleteSetup, saveSetups };
   182→}

   245→// ─── Auto-Calculator ─────────────────────────────────────────
   246→function calculateAuto(formData: TradeFormData) {
   247→  const entryPrice = parseFloat(formData.entryPrice);
   248→  const stopLoss = parseFloat(formData.stopLoss);
   249→  const takeProfit = parseFloat(formData.takeProfit);
   250→  const exitPrice = parseFloat(formData.exitPrice);
   251→  const lotSize = parseFloat(formData.lotSize);
   252→
   253→  const result: {
   254→    rr: number | null;
   255→    pnl: number | null;
   256→    resultLabel: "WIN" | "LOSS" | "BE" | null;
   257→    duration: string | null;
   258→  } = { rr: null, pnl: null, resultLabel: null, duration: null };
   259→
   260→  if (!isNaN(entryPrice) && !isNaN(stopLoss) && !isNaN(takeProfit)) {
   261→    const risk = Math.abs(entryPrice - stopLoss);
   262→    const reward = Math.abs(takeProfit - entryPrice);
   263→    if (risk > 0) {
   264→      result.rr = parseFloat((reward / risk).toFixed(2));
   265→    }
   266→  }
   267→
   268→  if (!isNaN(exitPrice) && !isNaN(entryPrice) && !isNaN(lotSize) && formData.exitPrice !== "") {
   269→    const priceDiff = formData.direction === "LONG"
   270→      ? exitPrice - entryPrice
   271→      : entryPrice - exitPrice;
   272→    const contractSize = getContractSize(formData.pair);
   273→    result.pnl = parseFloat((priceDiff * lotSize * contractSize).toFixed(2));
   274→  }
   275→
   276→  if (!isNaN(exitPrice) && !isNaN(entryPrice) && !isNaN(stopLoss) && !isNaN(takeProfit) && formData.exitPrice !== "") {
   277→    if (formData.direction === "LONG") {
   278→      if (exitPrice >= takeProfit) result.resultLabel = "WIN";
   279→      else if (exitPrice <= stopLoss) result.resultLabel = "LOSS";
   280→      else result.resultLabel = "BE";
   281→    } else {
   282→      if (exitPrice <= takeProfit) result.resultLabel = "WIN";
   283→      else if (exitPrice >= stopLoss) result.resultLabel = "LOSS";
   284→      else result.resultLabel = "BE";
   285→    }
   286→
   287→    // Adjust RR based on result to match backend logic
   288→    if (result.resultLabel === "LOSS") {
   289→      result.rr = -1;
   290→    } else if (result.resultLabel === "BE" && !isNaN(exitPrice) && formData.exitPrice !== "") {
   291→      // Calculate partial RR for BE: actual price movement / risk
   292→      const risk = Math.abs(entryPrice - stopLoss);
   293→      if (risk > 0) {
   294→        const priceDiff = formData.direction === "LONG"
   295→          ? exitPrice - entryPrice
   296→          : entryPrice - exitPrice;
   297→        result.rr = parseFloat((priceDiff / risk).toFixed(2));
   298→      }
   299→    }
   300→  }
   301→
   302→  if (formData.entryTime && formData.exitTime) {
   303→    const [eH, eM] = formData.entryTime.split(":").map(Number);
   304→    const [xH, xM] = formData.exitTime.split(":").map(Number);
   305→    if (!isNaN(eH) && !isNaN(eM) && !isNaN(xH) && !isNaN(xM)) {
   306→      let totalMinutes = (xH * 60 + xM) - (eH * 60 + eM);
   307→      if (totalMinutes < 0) totalMinutes += 24 * 60;
   308→      const hours = Math.floor(totalMinutes / 60);
   309→      const minutes = totalMinutes % 60;
   310→      result.duration = hours > 0 ? `${hours}h ${minutes}min` : `${minutes}min`;
   311→    }
   312→  }
   313→
   314→  return result;
   315→}

   805→// ─── Trade Verification List ─────────────────────────────────
   806→function TradeVerificationList({
   807→  trades,
   808→  language,
   809→  isAdmin,
   810→  onTradeClick,
   811→  onTradeDeleted,
   812→  loading,
   813→}: {
   814→  trades: any[];
   815→  language: "fr" | "en";
   816→  isAdmin: boolean;
   817→  onTradeClick: (id: string) => void;
   818→  onTradeDeleted: () => void;
   819→  loading: boolean;
   820→}) {
   821→  const [filters, setFilters] = useState({
   822→    condition: "",
   823→    setup: "",
   824→    entryModel: "",
   825→    structure: "",
   826→    timing: "",
   827→  });
   828→  const [showFilters, setShowFilters] = useState(false);
   829→
   830→  // Delete confirmation state
   831→  const [deleteTarget, setDeleteTarget] = useState<{ id: string; pair: string; direction: string } | null>(null);
   832→  const [isDeleting, setIsDeleting] = useState(false);
   833→
   834→  const handleDeleteTrade = useCallback(async () => {
   835→    if (!deleteTarget || isDeleting) return;
   836→    setIsDeleting(true);
   837→    try {
   838→      const res = await fetch(`/api/trades/${deleteTarget.id}`, { method: "DELETE" });
   839→      if (!res.ok) {
   840→        const data = await res.json();
   841→        throw new Error(data.error || "Delete failed");
   842→      }
   843→      toast.success(language === "fr" ? "Trade supprimé !" : "Trade deleted!");
   844→      setDeleteTarget(null);
   845→      onTradeDeleted();
   846→    } catch (error: any) {
   847→      toast.error(error.message || (language === "fr" ? "Erreur lors de la suppression" : "Delete failed"));
   848→    } finally {
   849→      setIsDeleting(false);
   850→    }
   851→  }, [deleteTarget, isDeleting, language, onTradeDeleted]);
   852→
   853→  // Stats
   854→  const longCount = trades.filter(t => t.direction === "LONG").length;
   855→  const shortCount = trades.filter(t => t.direction === "SHORT").length;
   856→  const totalRR = trades.reduce((s, t) => s + (t.rr ?? 0), 0);
   857→
   858→  // Apply filters
   859→  const { condition, setup, entryModel, structure, timing } = filters;
   860→  const filteredTrades = useMemo(() => {
   861→    return trades.filter(trade => {
   862→      if (condition && trade.marketCondition !== condition) return false;
   863→      if (setup && trade.setup !== setup) return false;
   864→      if (entryModel && trade.entryModel !== entryModel) return false;
   865→      if (structure && trade.structure !== structure) return false;
   866→      if (timing && trade.session !== timing) return false;
   867→      return true;
   868→    });
   869→  }, [trades, condition, setup, entryModel, structure, timing]);
   870→
   871→  return (
   872→    <div className="space-y-4">
   873→      {/* Stats bar */}
   874→      <Card className="p-4">
   875→        <div className="flex flex-wrap items-center gap-4">
   876→          <div className="flex items-center gap-2">
   877→            <BarChart3 className="w-4 h-4 text-muted-foreground" />
   878→            <span className="text-sm font-medium">{filteredTrades.length} trades</span>
   879→          </div>
   880→          <Badge variant="outline" className="text-xs font-mono">
   881→            RR Total: {totalRR.toFixed(2)}
   882→          </Badge>
   883→          <div className="flex items-center gap-1.5">
   884→            <Badge className="text-[10px] bg-profit/15 text-profit border-profit/20">
   885→              <ArrowUpRight className="w-3 h-3 mr-0.5" /> {longCount}
   886→            </Badge>
   887→            <Badge className="text-[10px] bg-loss/15 text-loss border-loss/20">
   888→              <ArrowDownRight className="w-3 h-3 mr-0.5" /> {shortCount}
   889→            </Badge>
   890→          </div>
   891→
   892→          <Button
   893→            variant="ghost"
   894→            size="sm"
   895→            className="ml-auto gap-1"
   896→            onClick={() => setShowFilters(!showFilters)}
   897→          >
   898→            <Filter className="w-3.5 h-3.5" />
   899→            {t(language, "filtres")}
   900→            <ChevronDown className={cn("w-3 h-3 transition-transform", showFilters && "rotate-180")} />
   901→          </Button>
   902→        </div>
   903→
   904→        {/* Filters */}
   905→        {showFilters && (
   906→          <div className="grid grid-cols-2 md:grid-cols-5 gap-3 mt-4 pt-4 border-t border-border">
   907→            <Select value={filters.condition} onValueChange={v => setFilters(p => ({ ...p, condition: v === "__all__" ? "" : v }))}>
   908→              <SelectTrigger className="h-8 text-xs"><SelectValue placeholder={t(language, "condition")} /></SelectTrigger>
   909→              <SelectContent>
   910→                <SelectItem value="__all__">{language === "fr" ? "Toutes" : "All"}</SelectItem>
   911→                {MARKET_CONDITIONS.map(mc => <SelectItem key={mc} value={mc}>{mc}</SelectItem>)}
   912→              </SelectContent>
   913→            </Select>
   914→            <Select value={filters.setup} onValueChange={v => setFilters(p => ({ ...p, setup: v === "__all__" ? "" : v }))}>
   915→              <SelectTrigger className="h-8 text-xs"><SelectValue placeholder={t(language, "setup")} /></SelectTrigger>
   916→              <SelectContent>
   917→                <SelectItem value="__all__">{language === "fr" ? "Tous" : "All"}</SelectItem>
   918→                {SETUPS.map(s => <SelectItem key={s} value={s}>{s}</SelectItem>)}
   919→              </SelectContent>
   920→            </Select>
   921→            <Select value={filters.entryModel} onValueChange={v => setFilters(p => ({ ...p, entryModel: v === "__all__" ? "" : v }))}>
   922→              <SelectTrigger className="h-8 text-xs"><SelectValue placeholder={t(language, "modelEntree")} /></SelectTrigger>
   923→              <SelectContent>
   924→                <SelectItem value="__all__">{language === "fr" ? "Tous" : "All"}</SelectItem>
   925→                {ENTRY_MODELS.map(em => <SelectItem key={em} value={em}>{em}</SelectItem>)}

   929→              <SelectTrigger className="h-8 text-xs"><SelectValue placeholder={t(language, "structure")} /></SelectTrigger>
   930→              <SelectContent>
   931→                <SelectItem value="__all__">{language === "fr" ? "Toutes" : "All"}</SelectItem>
   932→                {STRUCTURES.map(st => <SelectItem key={st} value={st}>{t(language, st.toLowerCase().replace(/è/g, "e").replace(/é/g, "e") as any) || st}</SelectItem>)}

   936→              <SelectTrigger className="h-8 text-xs"><SelectValue placeholder={t(language, "timing")} /></SelectTrigger>

   942→          </div>

   946→      {/* Trade list */}
   947→      <Card className="overflow-hidden">
   948→        {loading ? (
   949→          <div className="p-6 space-y-3">
   950→            {Array.from({ length: 5 }).map((_, i) => (
   951→              <Skeleton key={i} className="h-12 w-full" />
   952→            ))}
   953→          </div>
   954→        ) : filteredTrades.length === 0 ? (
   955→          <div className="p-12 text-center">
   956→            <BarChart3 className="w-12 h-12 mx-auto text-muted-foreground/30 mb-4" />
   957→            <p className="text-muted-foreground">{t(language, "noData")}</p>
   958→          </div>
   959→        ) : (
   960→          <>
   961→            {/* Desktop Table */}
   962→            <div className="hidden md:block">
   963→              <Table>
   964→                <TableHeader>
   965→                  <TableRow className="hover:bg-transparent">
   966→                    <TableHead className="text-xs font-semibold uppercase tracking-wider w-12">N°</TableHead>
   967→                    <TableHead className="text-xs font-semibold uppercase tracking-wider">{language === "fr" ? "Long/Short" : "Long/Short"}</TableHead>
