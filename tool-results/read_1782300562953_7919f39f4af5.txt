    93→// ─── Types ───────────────────────────────────────────────────
    94→interface DayTradeData {
    95→  date: Date;
    96→  trades: TradeData[];
    97→  totalRR: number;
    98→  totalPnL: number;
    99→  isProfit: boolean | null; // null = no trades
   100→}
   101→
   102→interface TradeData {
   103→  id: string;
   104→  userId: string;
   105→  date: string;
   106→  pair: string;
   107→  direction: string;
   108→  session: string;
   109→  marketCondition: string;
   110→  timeframe: string;
   111→  setup: string | null;
   112→  entryPrice: number;
   113→  stopLoss: number;
   114→  takeProfit: number;
   115→  exitPrice: number | null;
   116→  entryTime: string | null;
   117→  exitTime: string | null;
   118→  duration: string | null;
   119→  lotSize: number | null;
   120→  rr: number | null;
   121→  pnl: number | null;
   122→  result: string | null;
   123→  newsEnabled: boolean;
   124→  emotions: string | null;
   125→  confluence: string | null;
   126→  mistakes: string | null;
   127→  lessons: string | null;
   128→  notes: string | null;
   129→  createdAt: string;
   130→  screenshots: { id: string; type: string; url: string }[];
   131→}

   133→// ─── French Day Names ────────────────────────────────────────
   134→const FRENCH_DAYS = ["DIM", "LUN", "MAR", "MER", "JEU", "VEN", "SAM"];
   135→
   136→const FRENCH_MONTHS = [
   137→  "Janvier", "Février", "Mars", "Avril", "Mai", "Juin",
   138→  "Juillet", "Août", "Septembre", "Octobre", "Novembre", "Décembre",
   139→];
   140→
   141→const FRENCH_DAY_NAMES = [
   142→  "Dimanche", "Lundi", "Mardi", "Mercredi", "Jeudi", "Vendredi", "Samedi",
   143→];

   145→// ─── Main Component ──────────────────────────────────────────
   146→export function JournalTab() {
   147→  const { user, language, setScreenshotViewerUrl, setSelectedTradeId, setShowTradeDetail } = useAppStore();
   148→  const { period, setPeriod, filters } = useTimeFilter();
   149→  const { trades, loading, refetch } = useTrades(filters);
   150→
   151→  // Fetch on mount
   152→  React.useEffect(() => { refetch(); }, [refetch]);
   153→  const isMobile = useIsMobile();
   154→  const isAdminUser = user?.role === "admin";
   155→
   156→  const [currentMonth, setCurrentMonth] = useState(new Date());
   157→  const [selectedDate, setSelectedDate] = useState<Date | null>(null);
   158→  const [expandedTradeId, setExpandedTradeId] = useState<string | null>(null);
   159→  const [mobileDetailOpen, setMobileDetailOpen] = useState(false);
   160→
   161→  // Delete confirmation state
   162→  const [deleteTarget, setDeleteTarget] = useState<{ id: string; pair: string; direction: string } | null>(null);
   163→  const [isDeleting, setIsDeleting] = useState(false);
   164→
   165→  const handleDeleteTrade = useCallback(async () => {
   166→    if (!deleteTarget || isDeleting) return;
   167→    setIsDeleting(true);
   168→    try {
   169→      const res = await fetch(`/api/trades/${deleteTarget.id}`, { method: "DELETE" });
   170→      if (!res.ok) {
   171→        const data = await res.json();
   172→        throw new Error(data.error || "Delete failed");
   173→      }
   174→      toast.success(language === "fr" ? "Trade supprimé !" : "Trade deleted!");
   175→      setDeleteTarget(null);
   176→      refetch();
   177→    } catch (error: any) {
   178→      toast.error(error.message || (language === "fr" ? "Erreur lors de la suppression" : "Delete failed"));
   179→    } finally {
   180→      setIsDeleting(false);
   181→    }
   182→  }, [deleteTarget, isDeleting, language, refetch]);
   183→
   184→  // Group trades by date
   185→  const tradesByDate = useMemo(() => {
   186→    const map = new Map<string, DayTradeData>();
   187→
   188→    for (const trade of trades) {
   189→      const tradeDate = parseISO(trade.date);
   190→      if (!isValid(tradeDate)) continue;
   191→
   192→      const key = format(tradeDate, "yyyy-MM-dd");
   193→
   194→      if (!map.has(key)) {
   195→        map.set(key, {
   196→          date: tradeDate,
   197→          trades: [],
   198→          totalRR: 0,
   199→          totalPnL: 0,
   200→          isProfit: null,
   201→        });
   202→      }
   203→
   204→      const dayData = map.get(key)!;
   205→      dayData.trades.push(trade as TradeData);
   206→      if (trade.rr !== null) dayData.totalRR += trade.rr;
   207→      if (trade.pnl !== null) dayData.totalPnL += trade.pnl;
   208→    }
   209→
   210→    // Determine profit/loss for each day
   211→    for (const [, dayData] of map) {
   212→      if (dayData.trades.length > 0) {
   213→        dayData.isProfit = dayData.totalPnL >= 0;
   214→      }
   215→    }
   216→
   217→    return map;
   218→  }, [trades]);
   219→
   220→  // Calendar grid data
   221→  const calendarDays = useMemo(() => {
   222→    const monthStart = startOfMonth(currentMonth);
   223→    const monthEnd = endOfMonth(currentMonth);
   224→    const calendarStart = startOfWeek(monthStart, { weekStartsOn: 0 }); // Sunday
   225→    const calendarEnd = endOfWeek(monthEnd, { weekStartsOn: 0 });
   226→
   227→    return eachDayOfInterval({ start: calendarStart, end: calendarEnd });
   228→  }, [currentMonth]);
   229→
   230→  // Selected date trades
   231→  const selectedDayTrades = useMemo(() => {
   232→    if (!selectedDate) return null;
   233→    const key = format(selectedDate, "yyyy-MM-dd");
   234→    return tradesByDate.get(key) || null;
   235→  }, [selectedDate, tradesByDate]);
   236→
   237→  // Navigate months
   238→  const goToPreviousMonth = useCallback(() => {
   239→    setCurrentMonth((prev) => subMonths(prev, 1));
   240→  }, []);
   241→
   242→  const goToNextMonth = useCallback(() => {
   243→    setCurrentMonth((prev) => addMonths(prev, 1));
   244→  }, []);
   245→
   246→  // Handle date click
   247→  const handleDateClick = useCallback(
   248→    (date: Date) => {
   249→      const key = format(date, "yyyy-MM-dd");
   250→      const dayData = tradesByDate.get(key);
   251→      if (dayData && dayData.trades.length > 0) {
   252→        setSelectedDate(date);
   253→        setExpandedTradeId(null);
   254→        if (isMobile) {
   255→          setMobileDetailOpen(true);
   256→        }
   257→      }
   258→    },
   259→    [tradesByDate, isMobile]
   260→  );
   261→
   262→  // Format date header in French
   263→  const formatDateHeader = useCallback(
   264→    (date: Date) => {
   265→      const dayName = FRENCH_DAY_NAMES[getDay(date)];
   266→      const day = format(date, "d");
   267→      const month = FRENCH_MONTHS[date.getMonth()];
   268→      const year = date.getFullYear();
   269→      return `${dayName} ${day} ${month} ${year}`.toUpperCase();
   270→    },
   271→    []
   272→  );
   273→
   274→  // Handle trade click for detail view
   275→  const handleTradeDetailClick = useCallback(
   276→    (tradeId: string) => {
   277→      setSelectedTradeId(tradeId);
   278→      setShowTradeDetail(true);
   279→    },
   280→    [setSelectedTradeId, setShowTradeDetail]
   281→  );
   282→
   283→  // Trade detail panel content
   284→  const tradeDetailContent = useMemo(() => {
   285→    if (!selectedDate || !selectedDayTrades) {
   286→      return (
   287→        <div className="flex flex-col items-center justify-center h-full min-h-[300px] text-center p-6">
   288→          <CalendarIcon className="w-16 h-16 text-muted-foreground/20 mb-4" />
   289→          <p className="text-muted-foreground text-sm">
   290→            {t(language, "clickDate")}
   291→          </p>
   292→        </div>
   293→      );
   294→    }
   295→
   296→    const dayTotalRR = selectedDayTrades.totalRR;
   297→    const dayTotalPnL = selectedDayTrades.totalPnL;
   298→
   299→    return (
   300→      <div className="space-y-4">
   301→        {/* Date Header */}
   302→        <div className="space-y-2">
   303→          <h3 className="text-sm font-bold tracking-wider text-foreground/80">
   304→            {formatDateHeader(selectedDate)}
   305→          </h3>
   306→          <div className="flex items-center gap-3">
   307→            <Badge
   308→              variant="outline"
   309→              className={cn(
   310→                "text-xs font-mono gap-1",
   311→                dayTotalPnL >= 0
   312→                  ? "text-profit border-profit/30 bg-profit/5"
   313→                  : "text-loss border-loss/30 bg-loss/5"
   314→              )}
   315→            >
   316→              {dayTotalPnL >= 0 ? (
   317→                <TrendingUp className="w-3 h-3" />
   318→              ) : (
   319→                <TrendingDown className="w-3 h-3" />
   320→              )}
   321→              P&L: {dayTotalPnL >= 0 ? "+" : ""}
   322→              {dayTotalPnL.toFixed(2)}
   323→            </Badge>
   324→            <Badge variant="outline" className="text-xs font-mono gap-1">
   325→              <Target className="w-3 h-3" />
   326→              RR: {dayTotalRR.toFixed(2)}
   327→            </Badge>
   328→            <Badge variant="secondary" className="text-xs">
   329→              {selectedDayTrades.trades.length}{" "}
   330→              {selectedDayTrades.trades.length > 1
   331→                ? language === "fr"
   332→                  ? "trades"
   333→                  : "trades"
   334→                : "trade"}
   335→            </Badge>
   336→          </div>
   337→        </div>
   338→
   339→        <Separator />
   340→
   341→        {/* Trade Cards */}
   342→        <div className="space-y-3">
   343→          {selectedDayTrades.trades.map((trade, index) => (
   344→            <TradeCard
   345→              key={trade.id}
   346→              trade={trade}
   347→              index={index}
   348→              language={language}
   349→              isAdmin={isAdminUser}
   350→              isExpanded={expandedTradeId === trade.id}
   351→              onToggleExpand={() =>
   352→                setExpandedTradeId((prev) =>
   353→                  prev === trade.id ? null : trade.id
   354→                )
   355→              }
   356→              onScreenshotClick={setScreenshotViewerUrl}
   357→              onTradeDetailClick={handleTradeDetailClick}
   358→              onDeleteClick={(id) => setDeleteTarget({ id, pair: trade.pair, direction: trade.direction })}
   359→            />
   360→          ))}
   361→        </div>
   362→
   363→        {/* Mini Charts */}
   364→        {selectedDayTrades.trades.length > 1 && (
   365→          <>
   366→            <Separator />
   367→            <MiniCharts
   368→              trades={selectedDayTrades.trades}
   369→              language={language}
   370→            />
   371→          </>
   372→        )}
   373→      </div>
   374→    );
   375→  }, [
   376→    selectedDate,
   377→    selectedDayTrades,
   378→    language,
   379→    expandedTradeId,
   380→    formatDateHeader,
   381→    setScreenshotViewerUrl,
   382→    handleTradeDetailClick,
   383→    isAdminUser,
   384→    deleteTarget,
   385→    isDeleting,
   386→    handleDeleteTrade,
   387→  ]);
   388→
   389→  if (loading) {
   390→    return (
   391→      <div className="p-4 md:p-6 max-w-[1600px] mx-auto overflow-x-hidden">
   392→        <div className="flex flex-col lg:flex-row gap-6">
   393→          <div className="flex-1 lg:flex-[3]">
   394→            <Card className="p-6">
   395→              <Skeleton className="h-8 w-48 mb-6" />
   396→              <div className="grid grid-cols-7 gap-2">
   397→                {Array.from({ length: 35 }).map((_, i) => (
   398→                  <Skeleton key={i} className="h-16 w-full rounded-lg" />
   399→                ))}
   400→              </div>
   401→            </Card>
   402→          </div>
   403→          <div className="flex-1 lg:flex-[2]">
   404→            <Card className="p-6 h-80">
   405→              <Skeleton className="h-full w-full" />
   406→            </Card>
   407→          </div>
   408→        </div>
   409→      </div>
   410→    );
   411→  }
   412→
   413→  return (
   414→    <div className="p-4 md:p-6 max-w-[1600px] mx-auto overflow-x-hidden">
   415→      <div className="flex flex-col lg:flex-row gap-4 md:gap-6">
   416→        {/* Calendar Section (Left - 60%) */}
   417→        <div className="flex-1 lg:flex-[3]">
   418→          <Card className="overflow-hidden">
   419→            {/* Calendar Header */}
   420→            <div className="p-4 md:p-6 border-b border-border space-y-3">
   421→              <div className="flex items-center justify-between">
   422→                <h2 className="text-lg font-bold tracking-tight text-foreground">
   423→                  {FRENCH_MONTHS[currentMonth.getMonth()]}{" "}
   424→                  {currentMonth.getFullYear()}
   425→                </h2>
   426→                <div className="flex items-center gap-1">
   427→                  <Button
   428→                    variant="ghost"
   429→                    size="icon"
   430→                    onClick={goToPreviousMonth}
   431→                    className="h-8 w-8"
   432→                  >
   433→                    <ChevronLeft className="w-4 h-4" />
   434→                  </Button>
   435→                  <Button
   436→                    variant="ghost"
   437→                    size="icon"
   438→                    onClick={goToNextMonth}
   439→                    className="h-8 w-8"
   440→                  >
   441→                    <ChevronRight className="w-4 h-4" />
   442→                  </Button>
   443→                </div>
   444→              </div>
   445→              {/* Time Filter */}
   446→              <TimeFilterBar language={language} period={period} onPeriodChange={setPeriod} />
   447→            </div>
   448→
   449→            {/* Calendar Grid */}
   450→            <div className="p-3 md:p-4">
   451→              {/* Day Headers */}
   452→              <div className="grid grid-cols-7 mb-2">
   453→                {FRENCH_DAYS.map((day) => (
   454→                  <div
   455→                    key={day}
   456→                    className="text-center text-xs font-semibold text-muted-foreground uppercase tracking-wider py-1"
   457→                  >
   458→                    {day}
   459→                  </div>
   460→                ))}
   461→              </div>
   462→
   463→              {/* Day Cells */}
   464→              <div className="grid grid-cols-7 gap-1">
   465→                {calendarDays.map((day, idx) => {
   466→                  const key = format(day, "yyyy-MM-dd");
   467→                  const dayData = tradesByDate.get(key);
   468→                  const hasTrades = dayData && dayData.trades.length > 0;
   469→                  const isCurrentMonth = isSameMonth(day, currentMonth);
   470→                  const isSelected =
   471→                    selectedDate && isSameDay(day, selectedDate);
   472→                  const isTodayDate = isToday(day);
   473→
   474→                  return (
   475→                    <button
   476→                      key={idx}
   477→                      onClick={() => handleDateClick(day)}
   478→                      disabled={!hasTrades || !isCurrentMonth}
   479→                      className={cn(
   480→                        "relative flex flex-col items-center justify-center rounded-lg p-1 min-h-[44px] md:min-h-[56px] transition-all duration-200 text-sm",
   481→                        // Base styles
   482→                        "hover:bg-muted/50",
   483→                        // Current month vs outside
   484→                        isCurrentMonth
   485→                          ? "text-foreground"
   486→                          : "text-muted-foreground/40",
   487→                        // Today
   488→                        isTodayDate &&
   489→                          !isSelected &&
   490→                          "ring-1 ring-primary/30",
   491→                        // Selected
   492→                        isSelected &&
   493→                          "ring-2 ring-primary bg-primary/10",
   494→                        // Has trades - clickable
   495→                        hasTrades && isCurrentMonth
   496→                          ? "cursor-pointer hover:scale-[1.02]"
   497→                          : "cursor-default",
   498→                        // No trades - dim
   499→                        !hasTrades && isCurrentMonth && "opacity-60"
   500→                      )}
   501→                    >
   502→                      <span
   503→                        className={cn(
   504→                          "text-xs md:text-sm font-medium leading-none",
   505→                          isSelected && "text-primary font-bold"
   506→                        )}
   507→                      >
   508→                        {format(day, "d")}
   509→                      </span>
   510→                      {hasTrades && isCurrentMonth && (
   511→                        <>
   512→                          {/* Color indicator dot */}
   513→                          <div
   514→                            className={cn(
   515→                              "w-1.5 h-1.5 rounded-full mt-0.5",
   516→                              dayData!.isProfit
   517→                                ? "bg-profit"
   518→                                : "bg-loss"
   519→                            )}
   520→                          />
   521→                          {/* RR Value */}
   522→                          <span
   523→                            className={cn(
   524→                              "text-[9px] md:text-[10px] font-mono leading-none mt-0.5",
   525→                              dayData!.isProfit
   526→                                ? "text-profit"
   527→                                : "text-loss"
   528→                            )}
   529→                          >
   530→                            {dayData!.totalRR >= 0 ? "+" : ""}
   531→                            {dayData!.totalRR.toFixed(1)}
   532→                          </span>
   533→                        </>
   534→                      )}
   535→                    </button>
   536→                  );
   537→                })}
   538→              </div>
   539→            </div>
   540→
   541→            {/* Calendar Legend */}
   542→            <div className="px-4 pb-4 flex items-center gap-4 text-xs text-muted-foreground">
   543→              <div className="flex items-center gap-1.5">
   544→                <div className="w-2.5 h-2.5 rounded-full bg-profit" />
   545→                <span>{language === "fr" ? "Profit" : "Profit"}</span>
   546→              </div>
   547→              <div className="flex items-center gap-1.5">
   548→                <div className="w-2.5 h-2.5 rounded-full bg-loss" />
   549→                <span>{language === "fr" ? "Perte" : "Loss"}</span>
   550→              </div>
   551→              <div className="flex items-center gap-1.5">
   552→                <div className="w-2.5 h-2.5 rounded-full ring-1 ring-primary/30" />
   553→                <span>{language === "fr" ? "Aujourd'hui" : "Today"}</span>
   554→              </div>
   555→            </div>
   556→          </Card>
   557→        </div>
   558→
   559→        {/* Trade Detail Section (Right - 40%) */}
   560→        {!isMobile ? (
   561→          <div className="flex-1 lg:flex-[2]">
   562→            <Card className="h-full overflow-hidden">
   563→              <ScrollArea className="h-[calc(100vh-12rem)]">
   564→                <div className="p-4 md:p-6">{tradeDetailContent}</div>
   565→              </ScrollArea>
   566→            </Card>
   567→          </div>
   568→        ) : (
   569→          // Mobile: Sheet for trade detail
   570→          <Sheet open={mobileDetailOpen} onOpenChange={setMobileDetailOpen}>
   571→            <SheetContent side="bottom" className="h-[85vh] rounded-t-xl p-0">
   572→              <SheetHeader className="p-4 border-b border-border">
   573→                <SheetTitle className="text-sm">
   574→                  {selectedDate
   575→                    ? formatDateHeader(selectedDate)
   576→                    : t(language, "tradeDetails")}
   577→                </SheetTitle>
   578→              </SheetHeader>
   579→              <ScrollArea className="h-[calc(85vh-4rem)]">
   580→                <div className="p-4">{tradeDetailContent}</div>
   581→              </ScrollArea>
   582→            </SheetContent>
   583→          </Sheet>
   584→        )}
   585→      </div>
   586→
   587→      {/* Delete Confirmation Dialog */}
   588→      <Dialog open={!!deleteTarget} onOpenChange={(open) => { if (!open) setDeleteTarget(null); }}>
   589→        <DialogContent className="sm:max-w-md">
   590→          <DialogHeader>
   591→            <DialogTitle className="flex items-center gap-2">
   592→              <Trash2 className="w-5 h-5 text-destructive" />
   593→              {language === "fr" ? "Supprimer ce trade ?" : "Delete this trade?"}
   594→            </DialogTitle>
   595→            <DialogDescription>
   596→              {deleteTarget && (
   597→                language === "fr"
   598→                  ? `Voulez-vous supprimer le trade ${deleteTarget.direction} ${deleteTarget.pair} ? Cette action est irréversible.`
