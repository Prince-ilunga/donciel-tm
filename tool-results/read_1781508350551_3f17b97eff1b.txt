   258→/** Countdown timer display */
   259→function CountdownTimer({ targetDate, language }: { targetDate: string; language: string }) {
   260→  const [timeLeft, setTimeLeft] = useState("");
   261→  const isFr = language === "fr";
   262→
   263→  useEffect(() => {
   264→    const update = () => {
   265→      const now = new Date().getTime();
   266→      const target = new Date(targetDate).getTime();
   267→      const diff = target - now;
   268→
   269→      if (diff <= 0) {
   270→        setTimeLeft(isFr ? "En cours" : "Live");
   271→        return;
   272→      }
   273→
   274→      const hours = Math.floor(diff / (1000 * 60 * 60));
   275→      const minutes = Math.floor((diff % (1000 * 60 * 60)) / (1000 * 60));
   276→      const seconds = Math.floor((diff % (1000 * 60)) / 1000);
   277→
   278→      if (hours > 24) {
   279→        const days = Math.floor(hours / 24);
   280→        setTimeLeft(`${days}d ${hours % 24}h`);
   281→      } else if (hours > 0) {
   282→        setTimeLeft(`${hours}h ${minutes}m`);
   283→      } else {
   284→        setTimeLeft(`${minutes}m ${seconds}s`);
   285→      }
   286→    };
   287→
   288→    update();
   289→    const interval = setInterval(update, 1000);
   290→    return () => clearInterval(interval);
   291→  }, [targetDate, isFr]);
   292→
   293→  return <span className="text-[10px] font-mono font-bold text-primary">{timeLeft}</span>;
   294→}

   300→function CalendarSubTab({ language }: { language: string }) {
   301→  const isFr = language === "fr";
   302→  const [calendarData, setCalendarData] = useState<any>(null);
   303→  const [loading, setLoading] = useState(true);
   304→  const [error, setError] = useState<string | null>(null);
   305→  const [period, setPeriod] = useState<PeriodFilter>("today");
   306→  const [activeAsset, setActiveAsset] = useState<string | null>(null);
   307→  const intervalRef = useRef<NodeJS.Timeout | null>(null);
   308→
   309→  const fetchCalendar = useCallback(async () => {
   310→    try {
   311→      const assetParam = activeAsset ? `&asset=${activeAsset}` : '';
   312→      const res = await fetch(`/api/market/calendar?lang=${language}&period=${period}${assetParam}`);
   313→      if (!res.ok) throw new Error("Fetch failed");
   314→      const data = await res.json();
   315→      setCalendarData(data);
   316→      setError(null);
   317→    } catch {
   318→      setError(isFr ? "Erreur lors du chargement du calendrier" : "Error loading calendar");
   319→    } finally {
   320→      setLoading(false);
   321→    }
   322→  }, [language, isFr, period, activeAsset]);
   323→
   324→  // Reset loading when period or asset changes
   325→  useEffect(() => {
   326→    setLoading(true);
   327→  }, [period, activeAsset]);
   328→
   329→  useEffect(() => {
   330→    fetchCalendar();
   331→    intervalRef.current = setInterval(fetchCalendar, 5 * 60 * 1000);
   332→    return () => {
   333→      if (intervalRef.current) clearInterval(intervalRef.current);
   334→    };
   335→  }, [fetchCalendar]);
   336→
   337→  // Normalize events from API format to component format
   338→  const normalizedEvents = useMemo(() => {
   339→    const rawEvents: any[] = calendarData?.events || [];
   340→    return rawEvents.map((evt: any) => {
   341→      // Build a date from today + time
   342→      const today = new Date();
   343→      let eventDate: Date;
   344→      if (evt.date) {
   345→        eventDate = new Date(evt.date);
   346→      } else if (evt.time) {
   347→        const [h, m] = evt.time.split(":").map(Number);
   348→        eventDate = new Date(today.getFullYear(), today.getMonth(), today.getDate(), h || 0, m || 0);
   349→      } else {
   350→        eventDate = today;
   351→      }
   352→
   353→      return {
   354→        title: evt.event || evt.title || "",
   355→        date: eventDate.toISOString(),
   356→        time: evt.time || "",
   357→        currency: (evt.currency || "").toUpperCase(),
   358→        impact: (evt.impact || "low").toUpperCase(),
   359→        countryFlag: evt.country || "🌍",
   360→        forecast: evt.forecast || null,
   361→        previous: evt.previous || null,
   362→        actual: evt.actual || null,
   363→        interpretation: evt.interpretation || null,
   364→        direction: evt.direction || null,
   365→        impactedPairs: evt.impactedPairs || [],
   366→      };
   367→    });
   368→  }, [calendarData]);
   369→
   370→  // Find next high-impact event
   371→  const nextHighImpact = useMemo(() => {
   372→    const now = new Date();
   373→    return normalizedEvents.find(e => e.impact === "HIGH" && new Date(e.date) > now) || null;
   374→  }, [normalizedEvents]);
   375→
   376→  // Week events: use normalizedEvents directly when period=week (API already returns correct period)
   377→  // For period=today fallback, filter by date range
   378→  const weekEvents = useMemo(() => {
   379→    if (period === "week") {
   380→      // API already returns the correct events for the week - just sort them
   381→      return [...normalizedEvents].sort((a: any, b: any) => new Date(a.date).getTime() - new Date(b.date).getTime());
   382→    }
   383→    // Fallback: filter by local date range (used when period=today but showing week view)
   384→    const now = new Date();
   385→    const dayOfWeek = now.getDay();
   386→    const mondayOffset = dayOfWeek === 0 ? -6 : 1 - dayOfWeek;
   387→    const monday = new Date(now.getFullYear(), now.getMonth(), now.getDate() + mondayOffset);
   388→    const friday = new Date(monday);
   389→    friday.setDate(monday.getDate() + 5);
   390→
   391→    return normalizedEvents.filter((e: any) => {
   392→      const d = new Date(e.date);
   393→      return d >= monday && d < friday;
   394→    }).sort((a: any, b: any) => new Date(a.date).getTime() - new Date(b.date).getTime());
   395→  }, [normalizedEvents, period]);
   396→
   397→  // Today's events
   398→  const todayEvents = useMemo(() => {
   399→    const now = new Date();
   400→    return normalizedEvents.filter((e: any) => {
   401→      const d = new Date(e.date);
   402→      return d.toDateString() === now.toDateString();
   403→    });
   404→  }, [normalizedEvents]);
   405→
   406→  // Merge assetEvents + general events when asset is selected
   407→  const displayEvents = useMemo(() => {
   408→    if (activeAsset && calendarData?.assetEvents?.length > 0) {
   409→      // Normalize asset events the same way as general events
   410→      const today = new Date();
   411→      const assetEvts = calendarData.assetEvents.map((evt: any) => {
   412→        let eventDate: Date;
   413→        if (evt.date) {
   414→          eventDate = new Date(evt.date);
   415→        } else if (evt.time) {
   416→          const [h, m] = (evt.time || "").split(":").map(Number);
   417→          eventDate = new Date(today.getFullYear(), today.getMonth(), today.getDate(), h || 0, m || 0);
   418→        } else {
   419→          eventDate = today;
   420→        }
   421→        return {
   422→          title: evt.event || evt.title || "",
   423→          date: eventDate.toISOString(),
   424→          time: evt.time || "",
   425→          currency: (evt.currency || "").toUpperCase(),
   426→          impact: (evt.impact || "low").toUpperCase(),
   427→          countryFlag: evt.country || "🌍",
   428→          forecast: evt.forecast || null,
   429→          previous: evt.previous || null,
   430→          actual: evt.actual || null,
   431→          interpretation: evt.interpretation || null,
   432→          direction: evt.direction || null,
   433→          impactedPairs: evt.impactedPairs || [],
   434→        };
   435→      });
   436→      const generalEvts = normalizedEvents.filter((e: any) =>
   437→        !assetEvts.some((ae: any) => ae.title === e.title)
   438→      );
   439→      return { assetEvents: assetEvts, generalEvents: generalEvts };
   440→    }
   441→    return { assetEvents: [], generalEvents: normalizedEvents };
   442→  }, [activeAsset, calendarData, normalizedEvents]);
   443→
   444→  if (loading) {
   445→    return (
   446→      <div className="space-y-4">
   447→        <div className="grid grid-cols-3 gap-3">
   448→          <Card className="p-3 sm:p-4"><Skeleton className="h-8 w-12 mx-auto mb-1" /><Skeleton className="h-3 w-16 mx-auto" /></Card>
   449→          <Card className="p-3 sm:p-4"><Skeleton className="h-8 w-12 mx-auto mb-1" /><Skeleton className="h-3 w-16 mx-auto" /></Card>
   450→          <Card className="p-3 sm:p-4"><Skeleton className="h-8 w-12 mx-auto mb-1" /><Skeleton className="h-3 w-16 mx-auto" /></Card>
   451→        </div>
   452→        <Card className="p-4 sm:p-6">
   453→          <Skeleton className="h-6 w-48 mb-4" />
   454→          <div className="space-y-3">
   455→            {Array.from({ length: 6 }).map((_, i) => (
   456→              <div key={i} className="flex items-center gap-3">
   457→                <Skeleton className="h-4 w-14" />
   458→                <Skeleton className="h-4 w-6" />
   459→                <Skeleton className="h-4 w-10" />
   460→                <Skeleton className="h-4 flex-1" />
   461→              </div>
   462→            ))}
   463→          </div>
   464→        </Card>
   465→      </div>
   466→    );
   467→  }
   468→
   469→  if (error) {
   470→    return (
   471→      <Card className="p-12 text-center">
   472→        <AlertTriangle className="w-12 h-12 mx-auto text-muted-foreground/30 mb-4" />
   473→        <p className="text-muted-foreground">{error}</p>
   474→        <Button onClick={fetchCalendar} variant="outline" className="mt-4 gap-2">
   475→          <RefreshCw className="w-4 h-4" />
   476→          {isFr ? "Réessayer" : "Retry"}
   477→        </Button>
   478→      </Card>
   479→    );
   480→  }
   481→
   482→  const highCount = normalizedEvents.filter((e: any) => e.impact === "HIGH").length;
   483→  const mediumCount = normalizedEvents.filter((e: any) => e.impact === "MEDIUM").length;
