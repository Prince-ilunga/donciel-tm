    54→    54→// ─── Alert Hook ──────────────────────────────────────────
    55→    55→function useAlerts() {
    56→    56→  const [alerts, setAlerts] = useState<any[]>([]);
    57→    57→  const [loading, setLoading] = useState(true);
    58→    58→
    59→    59→  const fetchAlerts = useCallback(async () => {
    60→    60→    setLoading(true);
    61→    61→    try {
    62→    62→      const res = await fetch("/api/alerts");
    63→    63→      if (res.ok) {
    64→    64→        const data = await res.json();
    65→    65→        setAlerts(data.alerts || []);
    66→    66→      }
    67→    67→    } catch {
    68→    68→      setAlerts([]);
    69→    69→    }
    70→    70→    setLoading(false);
    71→    71→  }, []);
    72→    72→
    73→    73→  return { alerts, loading, refetch: fetchAlerts };
    74→    74→}

    76→    76→// ─── Notification Checker ────────────────────────────────
    77→    77→function useNotificationChecker(alerts: any[]) {
    78→    78→  const intervalRef = useRef<NodeJS.Timeout | null>(null);
    79→    79→  const triggeredRef = useRef<Set<string>>(new Set());
    80→    80→
    81→    81→  useEffect(() => {
    82→    82→    if (!alerts.length) return;
    83→    83→
    84→    84→    const checkAlerts = () => {
    85→    85→      const now = new Date();
    86→    86→      alerts.forEach((alert) => {
    87→    87→        if (alert.triggered) return;
    88→    88→        if (triggeredRef.current.has(alert.id)) return;
    89→    89→
    90→    90→        const alertTime = new Date(alert.alertDate);
    91→    91→        if (now >= alertTime) {
    92→    92→          triggeredRef.current.add(alert.id);
    93→    93→
    94→    94→          // Mark as triggered on server
    95→    95→          fetch(`/api/alerts/${alert.id}`, {
    96→    96→            method: "PUT",
    97→    97→            headers: { "Content-Type": "application/json" },
    98→    98→            body: JSON.stringify({ triggered: true }),
    99→    99→          }).catch(() => {});
   100→   100→
   101→   101→          // Show browser notification
   102→   102→          if ("Notification" in window) {
   103→   103→            if (Notification.permission === "granted") {
   104→   104→              new Notification("DONCIEL TM — Alerte", {
   105→   105→                body: alert.title + (alert.description ? `\n${alert.description}` : ""),
   106→   106→                icon: "/favicon.ico",
   107→   107→                tag: alert.id,
   108→   108→              });
   109→   109→            } else if (Notification.permission !== "denied") {
   110→   110→              Notification.requestPermission().then((perm) => {
   111→   111→                if (perm === "granted") {
   112→   112→                  new Notification("DONCIEL TM — Alerte", {
   113→   113→                    body: alert.title + (alert.description ? `\n${alert.description}` : ""),
   114→   114→                    icon: "/favicon.ico",
   115→   115→                    tag: alert.id,
   116→   116→                  });
   117→   117→                }
   118→   118→              });
   119→   119→            }
   120→   120→          }
   121→   121→        }
   122→   122→      });
   123→   123→    };
   124→   124→
   125→   125→    checkAlerts();
   126→   126→    intervalRef.current = setInterval(checkAlerts, 15000);
   127→   127→
   128→   128→    return () => {
   129→   129→      if (intervalRef.current) clearInterval(intervalRef.current);
   130→   130→    };
   131→   131→  }, [alerts]);
   132→   132→}
